"""
وحدة النحو — Grammar (Arabic/English).

المرتكزات الستة (verb, noun, prep, part, conj, kana) مع قوائم

module Grammar

using LinearAlgebra
using ..Physics.SyntaxFieldFull: SYNTAX_ANCHORS, KANA_VERBS, JAR_PREPS_LIST,
      CONJUNCTIONS_LIST, NEGATIONS_LIST, SyntaxFieldCache,
      compute_syntax_vector as syntax_field_compute_syntax_vector,
      expected_syntax as syntax_field_expected_syntax

export SYNTAX_ANCHORS, KANA_VERBS, JAR_PREPS_LIST, CONJUNCTIONS_LIST,
       NEGATIONS_LIST, compute_syntax_vector, expected_syntax,

"""
    compute_syntax_vector(word::String) -> Vector{Float64}

مرتكز نحوي 6D مطبَّع.
"""
function compute_syntax_vector(word::String)
    return syntax_field_compute_syntax_vector(word)
end

"""
    expected_syntax(prev_word, vocab, K) -> Vector{Float64}

النحو المتوقع بعد كلمة.

end

end # module Grammar
