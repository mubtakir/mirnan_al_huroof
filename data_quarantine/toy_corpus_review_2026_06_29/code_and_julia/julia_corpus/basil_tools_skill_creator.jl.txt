module SkillCreator

export SkillCreatorManager, create_skill

mutable struct SkillCreatorManager
    workspace_root::String
    agent_instance::Any
end

function SkillCreatorManager(workspace_root::String)
    return SkillCreatorManager(workspace_root, nothing)
end

function create_skill(manager::SkillCreatorManager, skill_name::String, class_name::String, method_name::String, code_content::String, description::String)::String
    file_name = "$(lowercase(replace(skill_name, "-" => "_"))).jl"
    tools_dir = joinpath(manager.workspace_root, "tools")
    mkpath(tools_dir)
    file_path = joinpath(tools_dir, file_name)
    
    if isfile(file_path)
        return "[ERROR] المهارة '$skill_name' موجودة بالفعل في $file_path. استخدم اسماً مختلفاً."
    end
    
    try
        full_code = """
        # Auto-generated Sovereign Skill: $skill_name
        # Description: $description
        # Generated by SkillCreator — BasilAgent.jl
        
        $code_content
        """
        write(file_path, full_code)
        @info "[SKILL] File written: $file_path"
        
        Base.include(Main, file_path)
        
        hot_registered = "تم التجميع بنجاح، لكن ToolRouter غير متوفر للتسجيل الفوري."
        if manager.agent_instance !== nothing && hasproperty(manager.agent_instance, :tools)
            router = manager.agent_instance.tools
            func_sym = Symbol(method_name)
            if isdefined(Main, func_sym)
                func_obj = getfield(Main, func_sym)
                from_router_func = (args...) -> func_obj(args...)
                router.tools[skill_name] = Main.ToolRouter.ToolMeta(skill_name, from_router_func, "evolution", false)
                hot_registered = "نجح ✅ — الأداة نشطة الآن في هذه الجلسة وتعمل فوراً."
            else
                hot_registered = "تم تجميع الملف، ولكن لم نجد الدالة '$method_name' مسجلة في النطاق العام Main."
            end
        end
        
        router_path = joinpath(manager.workspace_root, "src", "tool_router.jl")
        if isfile(router_path)
            content = read(router_path, String)
            target_comment = "function Router(workspace_root::String"
            if occursin(target_comment, content) && !occursin(skill_name, content)
                reg_block = """
                    # Dynamic Skill: $skill_name
                    try
                        Base.include(Main, joinpath(workspace_root, "tools", "$file_name"))
                        if isdefined(Main, :$method_name)
                            register!(router, "$skill_name", (args...) -> getfield(Main, :$method_name)(args...), "evolution", is_safe=false)
                        end
                    catch e
                        @warn "Failed to auto-load dynamic skill $skill_name: \\\$e"
                    end
                """
                search_anchor = "router = Router(workspace_root, Dict{String, ToolMeta}(), agent_instance)"
                if occursin(search_anchor, content)
                    content = replace(content, search_anchor => search_anchor * "\n" * reg_block)
                    write(router_path, content)
                    @info "[SKILL] Permanently registered in tool_router.jl"
                end
            end
        end
        
        status_parts = [
            "[SUCCESS] ✅ المهارة '$skill_name' أُنشئت وسُجِّلت بنجاح.",
            "  📁 الملف: $file_path",
            "  🔌 التسجيل الفوري: $hot_registered",
            "  ♾️ التسجيل الدائم: مُضاف لـ tool_router.jl (يصبح نشطاً من الجلسة القادمة تلقائياً)."
        ]
        return join(status_parts, "\n")
    catch e
        if isfile(file_path)
            rm(file_path)
        end
        return "[ERROR] فشل إنشاء المهارة: $e"
    end
end

end # module SkillCreator
