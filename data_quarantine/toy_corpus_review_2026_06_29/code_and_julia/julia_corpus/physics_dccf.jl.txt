"""
حقل الاقتران السياقي الديناميكي — DCCF (Dynamic Contextual Coupling Field).

بديل فيزيائي لـ self-attention.
يوزن العلاقات الطويلة المدى طورياً باضمحلال المسافة.
"""
module DCCF

using LinearAlgebra
using ..Constants: TOTAL_DIM
using ..WordPhysics: compute_extended_phase_vector

export DynamicCCF, get_context_boost, build_coupling

"""
    DynamicCCF

حقل اقتران ديناميكي — بديل self-attention.

الحقول:
- `decay_rate`: سرعة اضمحلال تأثير الكلمات البعيدة
- `mass_threshold`: عتبة التوافق الطوري الأدنى
"""
mutable struct DynamicCCF
    decay_rate::Float64
    mass_threshold::Float64
    pv_cache::Dict{String,Vector{Float64}}

    function DynamicCCF(; decay_rate::Float64=0.5, mass_threshold::Float64=0.3)
        return new(decay_rate, mass_threshold, Dict{String,Vector{Float64}}())
    end
end

function _get_pv(dccf::DynamicCCF, word::String)
    if !haskey(dccf.pv_cache, word)
        dccf.pv_cache[word] = Float64.(compute_extended_phase_vector(word))
    end
    return dccf.pv_cache[word]
end

"""
    build_coupling(dccf::DynamicCCF, context_words) -> Tuple{Matrix{Float64}, Dict}

بناء مصفوفة اقتران لحظية من سياق التوليد.

ترجع (coupling_matrix, scores_dict).
"""
function build_coupling(dccf::DynamicCCF, context_words::Vector{String})
    n = length(context_words)
    if n < 2
        return Matrix{Float64}(I, 1, 1), Dict()
    end

    pvs = [_get_pv(dccf, w) for w in context_words]
    coupling = zeros(Float64, n, n)
    scores = Dict()

    for i in 1:n
        for j in (i+1):n
            norm_i = norm(pvs[i])
            norm_j = norm(pvs[j])
            if norm_i < 1e-10 || norm_j < 1e-10
                continue
            end
            phase_align = dot(pvs[i], pvs[j]) / (norm_i * norm_j)

            mass_sim = norm_i * norm_j
            dist_decay = exp(-dccf.decay_rate * (j - i))

            val = phase_align * mass_sim * dist_decay
            coupling[i, j] = val
            coupling[j, i] = val
            scores[(i, j)] = Dict(
                "phase_align" => phase_align,
                "mass_sim" => mass_sim,
                "dist_decay" => dist_decay,
                "total" => val,
            )
        end
    end

    return coupling, scores
end

"""
    get_context_boost(dccf::DynamicCCF, word, context_words) -> Float64

حساب تعزيز DCCF لكلمة مرشحة ضد سياق التوليد.
"""
function get_context_boost(dccf::DynamicCCF, word::String,
                           context_words::Vector{String})
    n = length(context_words)
    if n < 1
        return 0.0
    end

    pw = _get_pv(dccf, word)
    norm_w = norm(pw)
    if norm_w < 1e-10
        return 0.0
    end

    total = 0.0
    weight_sum = 0.0
    for i in 1:n
        cpv = _get_pv(dccf, context_words[i])
        norm_c = norm(cpv)
        if norm_c < 1e-10
            continue
        end
        phase_align = dot(pw, cpv) / (norm_w * norm_c)
        if phase_align < dccf.mass_threshold
            continue
        end
        dist_decay = exp(-dccf.decay_rate * (n - i + 1))
        w = dist_decay * norm_c
        total += phase_align * w
        weight_sum += w
    end

    return total / max(weight_sum, 1e-10)
end

end # module DCCF
