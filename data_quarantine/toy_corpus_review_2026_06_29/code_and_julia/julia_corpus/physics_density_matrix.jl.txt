"""
مصفوفة الكثافة الطورية — Phase Density Matrix.

تمثيل السياق كحالة كمومية مختلطة (Mixed State):
  ρ = Σᵢ pᵢ |ψᵢ⟩⟨ψᵢ|

بديل فيزيائي لمتوسط المتجهات (الذي يفقد العلاقات البينية).
عند تقييم مرشح: R(v) = ⟨v| ρ |v⟩
"""
module DensityMatrix

using LinearAlgebra

export PhaseDensityMatrix, build!, resonance, get_trace, get_purity, get_spectral_entropy

"""
    PhaseDensityMatrix

ممثل السياق كمصفوفة كثافة طورية.

الحقول:
- `dim`: أبعاد فضاء الطور
- `decay_rate`: سرعة اضمحلال الأهمية الموضعية
- `rho`: مصفوفة الكثافة الحالية
"""
mutable struct PhaseDensityMatrix
    dim::Int
    decay_rate::Float64
    rho::Union{Matrix{Float64},Nothing}
    n_words::Int
    word_count::Int

    function PhaseDensityMatrix(; dim::Int=64, decay_rate::Float64=0.8)
        return new(dim, decay_rate, nothing, 0, 0)
    end
end

"""
    build!(dm::PhaseDensityMatrix, pv_list; dims=nothing) -> Matrix{Float64}

بناء مصفوفة الكثافة من قائمة متجهات طورية.

ρ = Σᵢ wᵢ · |ψᵢ⟩⟨ψᵢ|
حيث الأوزان تتبع اضمحلالاً أُسياً حسب الموضع.
"""
function build!(dm::PhaseDensityMatrix, pv_list::Vector{<:AbstractVector};
                dims::Union{Int,Tuple{Int,Int},Nothing}=nothing)
    if dims !== nothing
        if dims isa Tuple
            pvs = [Float64.(pv[dims[1]:dims[2]]) for pv in pv_list]
            d = dims[2] - dims[1] + 1
        else
            pvs = [Float64.(pv[1:dims]) for pv in pv_list]
            d = dims
        end
    else
        pvs = [Float64.(pv) for pv in pv_list]
        d = dm.dim
    end

    n = length(pvs)
    if n == 0
        dm.rho = zeros(d, d)
        dm.n_words = 0
        return dm.rho
    end

    weights = exp.(-dm.decay_rate .* Float64.(0:n-1))
    weights = reverse!(weights)
    weights ./= sum(weights)

    rho = zeros(Float64, d, d)
    for i in 1:n
        v = pvs[i][1:min(d, end)]
        v_norm = norm(v)
        if v_norm > 1e-10
            v ./= v_norm
        end
        rho .+= weights[i] .* (v * v')
    end

    dm.rho = rho
    dm.n_words = n
    dm.word_count = n
    return dm.rho
end

"""
    resonance(dm::PhaseDensityMatrix, candidate_pv; dims=nothing) -> Float64

قياس رنين مرشح: R = ⟨v| ρ |v⟩
"""
function resonance(dm::PhaseDensityMatrix, candidate_pv::AbstractVector;
                   dims::Union{Int,Tuple{Int,Int},Nothing}=nothing)
    if dm.rho === nothing || dm.n_words == 0
        return 0.0
    end

    if dims !== nothing
        if dims isa Tuple
            sub_rho = dm.rho[dims[1]:dims[2], dims[1]:dims[2]]
            v = Float64.(candidate_pv[dims[1]:dims[2]])
        else
            sub_rho = dm.rho[1:dims, 1:dims]
            v = Float64.(candidate_pv[1:dims])
        end
    else
        sub_rho = dm.rho
        v = Float64.(candidate_pv)
    end

    v_norm = norm(v)
    if v_norm > 1e-10
        v ./= v_norm
    end

    d = size(sub_rho, 1)
    v = v[1:d]
    resonance_val = v' * sub_rho * v
    return clamp(resonance_val, 0.0, 1.0)
end

"""
    get_trace(dm::PhaseDensityMatrix) -> Float64

أثر المصفوفة — إجمالي "الطاقة" في الحالة.
"""
function get_trace(dm::PhaseDensityMatrix)
    dm.rho === nothing && return 0.0
    return tr(dm.rho)
end

"""
    get_purity(dm::PhaseDensityMatrix) -> Float64

درجة النقاء Tr(ρ²) — 1.0 = حالة نقية.
"""
function get_purity(dm::PhaseDensityMatrix)
    dm.rho === nothing && return 0.0
    return tr(dm.rho * dm.rho)
end

"""
    get_spectral_entropy(dm::PhaseDensityMatrix) -> Float64

إنتروبيا فون نيومان الطيفية: S = -Tr(ρ ln ρ)
"""
function get_spectral_entropy(dm::PhaseDensityMatrix)
    if dm.rho === nothing || dm.n_words == 0
        return 0.0
    end
    eigenvals = eigvals(Hermitian(dm.rho))
    eigenvals = max.(eigenvals, 1e-12)
    return -sum(eigenvals .* log.(eigenvals))
end

end # module DensityMatrix
