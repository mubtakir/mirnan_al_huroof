- ربط الأحداث بالأماكن والأزمنة (Event Anchoring).

# 1. هياكل البيانات المتقدمة (Advanced Data Structures)

@dataclass
class TemporalInfo:
    """معلومة زمانية مرتبطة بحدث أو كيان."""
    expression: str          # النص الأصلي: "صباحاً"، "في العصر العباسي"
    normalized: str          # صيغة موحدة: "morning"، "750-1258 CE"
    type: str                # "time_of_day", "historical_period", "date", "duration"
    precision: str           # "exact", "approximate", "range"

@dataclass
class BinaryRelation:
    """علاقة ثنائية بين كيانين."""
    source: str              # الكيان الأول
    target: str              # الكيان الثاني
    relation_type: str       # نوع العلاقة: "parent_of", "teacher_of", "owns", "located_in"
    evidence: str            # الجملة التي استُنبطت منها
    confidence: float        # درجة الثقة (0.0 - 1.0)

@dataclass
class Event:
    """حدث مرتبط بزمان ومكان وفاعل."""
    description: str
    actor: Optional[str] = None
    location: Optional[str] = None
    time: Optional[TemporalInfo] = None
    participants: List[str] = field(default_factory=list)

@dataclass
class EntityKnowledge:
    """نسخة متطورة من معرفة الكيان."""
    name: str
    # --- طبقات المعرفة ---
    attributes: List[str] = field(default_factory=list)       # كريم، شجاع
    capabilities: List[str] = field(default_factory=list)     # يعمل، يصنع
    roles: List[str] = field(default_factory=list)            # طبيب، قاضي، أب
    workplace: Optional[str] = None                           # مستشفى، محكمة
    locations: List[str] = field(default_factory=list)        # كل الأماكن
    time_slots: List[TemporalInfo] = field(default_factory=list)  # الأزمنة المرتبطة
    events: List[Event] = field(default_factory=list)         # الأحداث المرتبطة
    relations_as_source: List[BinaryRelation] = field(default_factory=list)  # علاقات هو طرف أول فيها
    relations_as_target: List[BinaryRelation] = field(default_factory=list)  # علاقات هو طرف ثانٍ فيها
    inferred_relations: List[str] = field(default_factory=list)  # علاقات مستنبطة نصية

class AdvancedDiscoursePattern:
    """نمط خطابي متقدم مع دعم الاستخراج المنظم."""
    def __init__(self, name: str, regex_pattern: str, inference_type: str, 
                 sub_type: str = "", confidence: float = 0.9):
        self.name = name
        self.regex = re.compile(regex_pattern, re.IGNORECASE | re.VERBOSE)
        self.inference_type = inference_type
        self.sub_type = sub_type
        self.confidence = confidence
    
    def extract(self, sentence: str) -> Optional[Dict]:
        """محاولة مطابقة النمط مع الجملة."""
        match = self.regex.search(sentence)
        if match:
            result = match.groupdict()
            result['_pattern_name'] = self.name
            result['_confidence'] = self.confidence
            return result
        return None

# 3. قاعدة المعرفة المتقدمة (Advanced Knowledge Base)
# ==============================================================================

class AdvancedKnowledgeBase:
    """قاعدة معرفة متطورة تدعم كل أنواع الاستنباط."""
    
    def __init__(self):
        self.entities: Dict[str, EntityKnowledge] = {}
        self.events: List[Event] = []
        self.relations: List[BinaryRelation] = []
        self.temporal_expressions: List[TemporalInfo] = []
        self.inference_log: List[Dict] = []
    
    def _clean_name(self, name: str) -> str:
        """تنظيف وتوحيد أسماء الكيانات."""
        if not name:
            return ""
        name = name.strip()
        # إزالة أل التعريف للتوحيد
        if name.startswith("ال") and len(name) > 3:
            name = name[2:]
        return name
    
    def get_or_create_entity(self, name: str) -> EntityKnowledge:
        """استرجاع كيان أو إنشاؤه."""
        clean = self._clean_name(name)
        if clean not in self.entities:
            self.entities[clean] = EntityKnowledge(name=name.strip())
        return self.entities[clean]
    
    # === دوال إضافة المعرفة الجديدة ===
    
    def add_temporal_info(self, entity_name: str, temporal: TemporalInfo, pattern_name: str):
        """إضافة معلومة زمانية لكيان."""
        entity = self.get_or_create_entity(entity_name)
        if temporal not in entity.time_slots:
            entity.time_slots.append(temporal)
            self.inference_log.append({
                "type": "temporal",
                "entity": self._clean_name(entity_name),
                "value": temporal.expression,
                "normalized": temporal.normalized,
                "pattern": pattern_name
            })
    
    def add_relation(self, source: str, target: str, relation_type: str, 
                     evidence: str, confidence: float):
        """إضافة علاقة ثنائية بين كيانين."""
        source_clean = self._clean_name(source)
        target_clean = self._clean_name(target)
        
        relation = BinaryRelation(
            source=source_clean,
            target=target_clean,
            relation_type=relation_type,
            evidence=evidence,
            confidence=confidence
        )
        
        self.relations.append(relation)
        
        # ربط الكيانين بالعلاقة
        source_entity = self.get_or_create_entity(source)
        target_entity = self.get_or_create_entity(target)
        source_entity.relations_as_source.append(relation)
        target_entity.relations_as_target.append(relation)
        
        self.inference_log.append({
            "type": "relation",
            "source": source_clean,
            "target": target_clean,
            "relation": relation_type,
            "confidence": confidence,
            "evidence": evidence
        })
    
    def add_role(self, entity_name: str, role: str, pattern_name: str):
        """إضافة دور أو مهنة لكيان."""
        entity = self.get_or_create_entity(entity_name)
        role_clean = role.strip()
        if role_clean not in entity.roles:
            entity.roles.append(role_clean)
            self.inference_log.append({
                "type": "role",
                "entity": self._clean_name(entity_name),
                "value": role_clean,
                "pattern": pattern_name
            })
    
    def add_event(self, event: Event):
        """إضافة حدث مع ارتباطاته."""
        self.events.append(event)
        if event.actor:
            actor = self.get_or_create_entity(event.actor)
            actor.events.append(event)
    
    # === دوال الاستعلام والعرض ===
    
    def get_entity_relations(self, entity_name: str) -> List[BinaryRelation]:
        """استرجاع كل علاقات كيان معين."""
        clean = self._clean_name(entity_name)
        entity = self.entities.get(clean)
        if entity:
            return entity.relations_as_source + entity.relations_as_target
        return []
    
    def get_entities_with_attribute(self, attribute: str) -> List[str]:
        """استرجاع الكيانات التي تمتلك صفة معينة."""
        result = []
        for name, entity in self.entities.items():
            if attribute in entity.attributes:
                result.append(name)
        return result
    
    def find_relations_between(self, entity1: str, entity2: str) -> List[BinaryRelation]:
        """البحث عن علاقات مباشرة بين كيانين."""
        c1, c2 = self._clean_name(entity1), self._clean_name(entity2)
        result = []
        for rel in self.relations:
            if (rel.source == c1 and rel.target == c2) or \
               (rel.source == c2 and rel.target == c1):
                result.append(rel)
        return result
    
    def display_entity_profile(self, entity_name: str) -> str:
        """عرض ملف كامل لكيان معين."""
        clean = self._clean_name(entity_name)
        entity = self.entities.get(clean)
        if not entity:
            return f"⚠️ الكيان '{entity_name}' غير موجود في القاعدة."
        
        lines = []
        lines.append(f"\n{'='*50}")
        lines.append(f"📋 ملف الكيان: {entity.name}")
        lines.append(f"{'='*50}")
        
        if entity.attributes:
            lines.append(f"  🏷️  الصفات: {', '.join(entity.attributes)}")
        if entity.roles:
            lines.append(f"  👔 الأدوار: {', '.join(entity.roles)}")
        if entity.capabilities:
            lines.append(f"  ⚡ القدرات: {', '.join(entity.capabilities)}")
        if entity.workplace:
            lines.append(f"  🏢 مكان العمل: {entity.workplace}")
        if entity.locations:
            lines.append(f"  📍 الأماكن: {', '.join(entity.locations)}")
        if entity.time_slots:
            times = [t.expression for t in entity.time_slots]
            lines.append(f"  🕐 الأزمنة: {', '.join(times)}")
        
        # العلاقات
        all_rels = entity.relations_as_source + entity.relations_as_target
        if all_rels:
            lines.append(f"  🔗 العلاقات ({len(all_rels)}):")
            for rel in all_rels:
                if rel.source == clean:
                    lines.append(f"     → {rel.relation_type}: {rel.target} (ثقة: {rel.confidence:.0%})")
                else:
                    lines.append(f"     ← {rel.relation_type}: {rel.source} (ثقة: {rel.confidence:.0%})")
        
        # العلاقات المستنبطة
        if entity.inferred_relations:
            lines.append(f"  💡 استنباطات:")
            for inf in entity.inferred_relations:
                lines.append(f"     • {inf}")
        
        return '\n'.join(lines)
    
    def display_full_report(self) -> str:
        """عرض تقرير كامل عن القاعدة المعرفية."""
        lines = []
        lines.append("\n" + "="*60)
        lines.append("📚 تقرير قاعدة المعرفة المتقدمة")
        lines.append("="*60)
        
        lines.append(f"\n🔢 إحصائيات:")
        lines.append(f"   • عدد الكيانات: {len(self.entities)}")
        lines.append(f"   • عدد العلاقات: {len(self.relations)}")
        lines.append(f"   • عدد الأحداث: {len(self.events)}")
        lines.append(f"   • عدد الاستنباطات: {len(self.inference_log)}")
        
        # عرض أبرز الكيانات
        lines.append(f"\n🌟 أبرز الكيانات:")
        for name, entity in sorted(self.entities.items(), 
                                   key=lambda x: len(x[1].attributes) + len(x[1].relations_as_source), 
                                   reverse=True)[:5]:
            attr_count = len(entity.attributes)
            rel_count = len(entity.relations_as_source) + len(entity.relations_as_target)
            lines.append(f"   • {entity.name} ({attr_count} صفات, {rel_count} علاقات)")
        
        # عرض أنواع العلاقات
        if self.relations:
            rel_types = set(r.relation_type for r in self.relations)
            lines.append(f"\n🔗 أنواع العلاقات المكتشفة:")
            for rt in sorted(rel_types):
                count = sum(1 for r in self.relations if r.relation_type == rt)
                lines.append(f"   • {rt}: {count}")
        
        # عرض الكيانات حسب الدور
        roles_map: Dict[str, List[str]] = {}
        for name, entity in self.entities.items():
            for role in entity.roles:
                if role not in roles_map:
                    roles_map[role] = []
                roles_map[role].append(entity.name)
        
        if roles_map:
            lines.append(f"\n👔 الكيانات حسب الأدوار:")
            for role, entities in sorted(roles_map.items()):
                lines.append(f"   • {role}: {', '.join(entities)}")
        
        return '\n'.join(lines)

# 4. المحرك المتقدم (Advanced Inference Engine)
# ==============================================================================

class AdvancedInferenceEngine:
    """المحرك الرئيسي الذي يدمج كل أنواع الاستنباط."""
    
    def __init__(self):
        self.kb = AdvancedKnowledgeBase()
        self.all_patterns = (TEMPORAL_PATTERNS + RELATIONAL_PATTERNS + ROLE_PATTERNS)
    
    def _normalize_temporal(self, expression: str, t_type: str) -> TemporalInfo:
        """تطبيع التعبير الزماني إلى صيغة موحدة."""
        normalized = expression
        
        if t_type == "time_of_day":
            time_map = {
                "الصباح": "morning", "صباحاً": "morning",
                "المساء": "evening", "مساءً": "evening",
                "الليل": "night", "ليلاً": "night",
                "الظهر": "noon", "ظهراً": "noon",
                "العصر": "afternoon", "عصراً": "afternoon",
                "الفجر": "dawn", "فجراً": "dawn",
            }
            normalized = time_map.get(expression, expression)
        
        elif t_type == "historical_period":
            period_map = {
                "العصر العباسي": "Abbasid Era (750-1258 CE)",
                "العصر الأموي": "Umayyad Era (661-750 CE)",
                "العصر العثماني": "Ottoman Era (1299-1922 CE)",
                "العصر الجاهلي": "Pre-Islamic Era",
            }
            normalized = period_map.get(expression, expression)
        
        return TemporalInfo(
            expression=expression,
            normalized=normalized,
            type=t_type,
            precision="approximate"
        )
    
    def _map_relation_type(self, sub_type: str) -> str:
        """تحويل نوع العلاقة الفرعي إلى نوع موحد."""
        mapping = {
            "kinship": "قرابة",
            "teacher_student": "تعليم",
            "ownership": "ملكية",
            "friendship": "صداقة",
            "enmity": "عداوة",
            "located_in": "موقع",
        }
        return mapping.get(sub_type, sub_type)
    
    def analyze_sentence(self, sentence: str) -> List[Dict]:
        """تحليل جملة واحدة بكل الأنماط المتاحة."""
        sentence = sentence.strip()
        results = []
        
        for pattern in self.all_patterns:
            extracted = pattern.extract(sentence)
            if not extracted:
                continue
            
            result = {
                "sentence": sentence,
                "pattern": pattern.name,
                "type": pattern.inference_type,
                "sub_type": pattern.sub_type,
                "confidence": pattern.confidence,
            }
            
            # === تطبيق الاستنباط حسب النوع ===
            
            if pattern.inference_type == "temporal":
                time_expr = extracted.get('time', '')
                if time_expr:
                    temporal_info = self._normalize_temporal(time_expr, pattern.sub_type)
                    entity = extracted.get('entity', 'الراوي')
                    if entity:
                        self.kb.add_temporal_info(entity, temporal_info, pattern.name)
                    result["action"] = f"زمن: '{time_expr}' ({temporal_info.normalized})"
            
            elif pattern.inference_type == "relational":
                entity1 = extracted.get('entity1', '')
                entity2 = extracted.get('entity2', '')
                if entity1 and entity2:
                    rel_type = self._map_relation_type(pattern.sub_type)
                    self.kb.add_relation(entity1, entity2, rel_type, 
                                         sentence, pattern.confidence)
                    result["action"] = f"علاقة {rel_type}: {entity1} ←→ {entity2}"
            
            elif pattern.inference_type == "role":
                entity = extracted.get('entity', '')
                role = extracted.get('role', '')
                if entity and role:
                    self.kb.add_role(entity, role, pattern.name)
                    result["action"] = f"دور: '{entity}' هو '{role}'"
            
            results.append(result)
        
        return results
    
    def analyze_sentences(self, sentences: List[str]) -> AdvancedKnowledgeBase:
        """تحليل قائمة جمل كاملة."""
        print("🔄 جارٍ تحليل الجمل (الزمان، العلاقات، الأدوار)...\n")
        
        for i, sentence in enumerate(sentences, 1):
            print(f"  [{i}] 📝 {sentence}")
            results = self.analyze_sentence(sentence)
            if results:
                for res in results:
                    icon = {"temporal": "🕐", "relational": "🔗", "role": "👔"}.get(res['type'], "📌")
                    print(f"       {icon} [{res['sub_type']}] {res['action']} (ثقة: {res['confidence']:.0%})")
            else:
                print(f"       ⚠️ لا يوجد استنباط متقدم.")
        
        return self.kb

    def segment_to_list(text: str) -> List[str]:
        sentences = re.split(r'(?<=[.!؟])\s+', text)
        return [s.strip() for s in sentences if s.strip()]

def deep_understand(text: str, use_advanced: bool = True) -> AdvancedKnowledgeBase:
    """
    واجهة الفهم العميق الكامل:
    1. تفكيك النص إلى جمل.
    2. استنباط أساسي (صفات، قدرات، أماكن).
    3. استنباط متقدم (أزمنة، علاقات، أدوار).
    """
    print("\n" + "="*60)
    print("🧠 نظام الفهم العميق - نموذج القلم 3.0")
    print("="*60)
    
    # الخطوة 1: التفكيك
    print("\n✂️ تفكيك النص...")
    sentences = segment_to_list(text)
    print(f"   ✅ {len(sentences)} جملة\n")
    
    # الخطوة 2: الاستنباط الأساسي (إذا كان متاحاً)
    if BasicInferenceEngine:
        print("📦 تشغيل الاستنباط الأساسي (صفات، قدرات، أماكن)...")
        basic_engine = BasicInferenceEngine()
        # نستخدم نفس قاعدة المعرفة المتقدمة
        for sentence in sentences:
            basic_engine.analyze_sentence(sentence)
        print("   ✅ اكتمل الاستنباط الأساسي\n")
    
    # الخطوة 3: الاستنباط المتقدم
    if use_advanced:
        print("🚀 تشغيل الاستنباط المتقدم (أزمنة، علاقات، أدوار)...")
        advanced_engine = AdvancedInferenceEngine()
        kb = advanced_engine.analyze_sentences(sentences)
    else:
        kb = AdvancedKnowledgeBase()
    
    # الخطوة 4: عرض التقرير
    print(kb.display_full_report())
    
    return kb
