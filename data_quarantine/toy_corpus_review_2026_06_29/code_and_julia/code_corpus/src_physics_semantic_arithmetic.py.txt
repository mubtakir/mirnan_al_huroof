كما أن الحساب العددي مبني على "الواحد" (1) كوحدة مسافة،

العمليات:
  - الجمع الدلالي:    كلمة + Δ = معنى جديد
  - الطرح الدلالي:    كلمة₂ - كلمة₁ = Δ
  - القياس/التناسب:   أ:ب :: جـ:؟
  - المزج:            كلمة₁ + كلمة₂ (بنسبة)

مُكيّف لمرنان الحروف (يستخدم LetterDB وليس SemanticPhysicsEngine).

# ذاكرة تخزين مؤقت
_word_mv_cache: Dict[str, Multivector22] = {}
_word_linear_cache: Dict[str, np.ndarray] = {}


def _get_db():
    from src.physics.letter_db import LetterDB
    return LetterDB()


def _get_modulated_letter_vectors(word: str, db) -> List[np.ndarray]:
    """تفكيك الكلمة العربية مع تطبيق تأثير الحركات بنصف قيمة حروف المد المقابلة في الفضاء 22D."""
    _NORMALIZE = str.maketrans({
        '\u0622': '\u0627',  # آ → ا
        '\u0623': '\u0627',  # أ → ا
        '\u0625': '\u0627',  # إ → ا
        '\u0624': '\u0648',  # ؤ → و
        '\u0626': '\u064A',  # ئ → ي
        '\u0629': '\u0647',  # ة → ه
        '\u0649': '\u064A',  # ى → ي
    })
    
    parsed = []
    current_letter = None
    current_harakat = []
    
    for ch in word:
        cp = ord(ch)
        if 0x0621 <= cp <= 0x064A or ch in _NORMALIZE:
            norm_ch = ch.translate(_NORMALIZE)
            if current_letter is not None:
                parsed.append((current_letter, current_harakat))
            current_letter = norm_ch
            current_harakat = []
        elif 0x064B <= cp <= 0x0652:
            if current_letter is not None:
                current_harakat.append(cp)
                
    if current_letter is not None:
        parsed.append((current_letter, current_harakat))
        
    vectors = []
    for letter, harakat in parsed:
        v = db.get_vector(letter).copy()
        for h in harakat:
            if h in (0x064E, 0x064B):  # الفتحة ➔ نصف الألف
                v += 0.5 * db.get_vector('ا')
            elif h in (0x064F, 0x064C):  # الضمة ➔ نصف الواو
                v += 0.5 * db.get_vector('و')
            elif h in (0x0650, 0x064D):  # الكسرة ➔ نصف الياء
                v += 0.5 * db.get_vector('ي')
            elif h == 0x0651:  # الشدة ➔ تضعيف الحرف
                v *= 2.0
        vectors.append(v)
    return vectors


def _get_word_mv(word: str) -> Multivector22:
    """استرجاع أو حساب المتجه الهندسي لكلمة مع معالجة التشكيل."""
    if word in _word_mv_cache:
        return _word_mv_cache[word]
    db = _get_db()
    letters = _get_modulated_letter_vectors(word, db)
    mv = geometric_word_fusion(letters)
    _word_mv_cache[word] = mv
    return mv


def _get_word_linear(word: str) -> np.ndarray:
    """استرجاع أو حساب المتجه الخطي لكلمة مع معالجة التشكيل."""
    if word in _word_linear_cache:
        return _word_linear_cache[word]
    db = _get_db()
    letters = _get_modulated_letter_vectors(word, db)
    if not letters:
        return np.zeros(22)
    
    position_weights = [3.5, 2.5, 2.0, 1.5, 1.2, 1.0, 0.8, 0.6, 0.4, 0.3]
    weighted = np.zeros(22)
    for i, v in enumerate(letters):
        w = position_weights[i] if i < len(position_weights) else position_weights[-1]
        weighted += v * w
    nrm = np.linalg.norm(weighted)
    v_lin = weighted / nrm if nrm > 1e-10 else weighted
    
    _word_linear_cache[word] = v_lin
    return v_lin

def letter_distance_matrix() -> np.ndarray:
    """مصفوفة المسافات الخطية بين كل الحروف (28×28)."""
    db = _get_db()
    n = len(ARABIC_LETTERS)
    vectors = [db.get_vector(ch) for ch in ARABIC_LETTERS]
    dist = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            dist[i][j] = np.linalg.norm(vectors[i] - vectors[j])
    return dist


def letter_difference_vector(letter_from: str, letter_to: str) -> np.ndarray:
    """متجه الفرق الخطي: Δ = v(إلى) - v(من)."""
    db = _get_db()
    return db.get_vector(letter_to) - db.get_vector(letter_from)


def find_fundamental_semantic_units() -> Dict:
    """اكتشاف 'وحدات المسافة الأساسية' بين الحروف."""
    dist = letter_distance_matrix()
    pairs = []
    n = len(ARABIC_LETTERS)
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((ARABIC_LETTERS[i], ARABIC_LETTERS[j], float(dist[i][j])))
    pairs.sort(key=lambda x: x[2])

    closest = pairs[:5]
    farthest = pairs[-5:]

    db = _get_db()
    all_letter_vectors = np.array([db.get_vector(ch) for ch in ARABIC_LETTERS])
    dim_variances = np.var(all_letter_vectors, axis=0)
    top_dims_idx = np.argsort(dim_variances)[::-1][:5]
    top_dims = [(DIM_NAMES[i], round(float(dim_variances[i]), 4)) for i in top_dims_idx]

    distances = [p[2] for p in pairs]

    return {
        "distance_mean": round(float(np.mean(distances)), 4),
        "distance_std": round(float(np.std(distances)), 4),
        "distance_min": round(float(np.min(distances)), 4),
        "distance_max": round(float(np.max(distances)), 4),
        "fundamental_unit_norm": round(float(np.min(distances)), 4),
        "closest_pairs": [(f, t, round(d, 4)) for f, t, d in closest],
        "farthest_pairs": [(f, t, round(d, 4)) for f, t, d in farthest],
        "dominant_dimensions": top_dims,
        "distance_histogram": {
            "0.0-0.5": sum(1 for d in distances if d < 0.5),
            "0.5-1.0": sum(1 for d in distances if 0.5 <= d < 1.0),
            "1.0-1.5": sum(1 for d in distances if 1.0 <= d < 1.5),
            "1.5-2.0": sum(1 for d in distances if 1.5 <= d < 2.0),
            "2.0-2.5": sum(1 for d in distances if 2.0 <= d < 2.5),
            "2.5+": sum(1 for d in distances if d >= 2.5),
        },
    }

def semantic_difference(word_from: str, word_to: str, method: str = "geometric") -> Dict:
    """حساب المسافة الدلالية بين كلمتين: Δ = كلمة₂ - كلمة₁."""
    import re

    if method == "linear":
        v_from = _get_word_linear(word_from)
        v_to = _get_word_linear(word_to)
        delta = v_to - v_from
        mag = float(np.linalg.norm(delta))
        top_idx = np.argsort(np.abs(delta))[::-1][:5]
        top_shifts = [(DIM_NAMES[i], round(float(delta[i]), 4)) for i in top_idx]
        return {
            "method": "linear", "word_from": word_from, "word_to": word_to,
            "delta_vector": [round(float(x), 4) for x in delta],
            "magnitude": round(mag, 4), "dominant_shift": top_shifts,
        }
    else:
        mv_from = _get_word_mv(word_from)
        mv_to = _get_word_mv(word_to)
        delta_mv = mv_to / mv_from
        return {
            "method": "geometric", "word_from": word_from, "word_to": word_to,
            "delta_multivector": str(delta_mv),
            "delta_scalar": round(float(delta_mv.s), 4),
            "delta_vector_norm": round(float(np.linalg.norm(delta_mv.v)), 4),
            "delta_bivector_norm": round(float(np.linalg.norm(delta_mv.b)), 4),
            "magnitude": round(float(delta_mv.norm()), 4),
        }


def semantic_midpoint(word1: str, word2: str) -> Dict:
    """إيجاد نقطة المنتصف الدلالي بين كلمتين."""
    mv1 = _get_word_mv(word1)
    mv2 = _get_word_mv(word2)
    midpoint = Multivector22(
        scalar=(mv1.s + mv2.s) / 2,
        vector=(mv1.v + mv2.v) / 2,
        bivector=(mv1.b + mv2.b) / 2,
    )
    midpoint.normalize()
    nearest_word, nearest_sim = _find_nearest_word(midpoint)
    return {
        "word1": word1, "word2": word2,
        "midpoint_mv": str(midpoint),
        "nearest_real_word": nearest_word,
        "nearest_similarity": round(float(nearest_sim), 4),
    }


def blend_words(word1: str, word2: str, ratio: float = 0.5) -> Dict:
    """مزج كلمتين بنسبة: blend = (1-ratio)*word1 + ratio*word2."""
    mv1 = _get_word_mv(word1)
    mv2 = _get_word_mv(word2)
    blend = Multivector22(
        scalar=mv1.s * (1 - ratio) + mv2.s * ratio,
        vector=mv1.v * (1 - ratio) + mv2.v * ratio,
        bivector=mv1.b * (1 - ratio) + mv2.b * ratio,
    )
    blend.normalize()
    nearest, sim = _find_nearest_word(blend)
    candidates = _find_top_k_words(blend, k=3)
    return {
        "word1": word1, "word2": word2, "ratio": ratio,
        "nearest_word": nearest,
        "similarity": round(float(sim), 4),
        "candidates": candidates,
    }

def solve_analogy(a: str, b: str, c: str, method: str = "linear") -> Dict:
    """حل التناسب: أ:ب :: جـ:؟"""
    import re

    if method == "linear":
        v_a = _get_word_linear(a)
        v_b = _get_word_linear(b)
        v_c = _get_word_linear(c)
        delta_v = v_b - v_a
        v_d = v_c + delta_v
        v_d_norm = v_d / (np.linalg.norm(v_d) + 1e-10)

        try:
            from model.vocab import ID2WORD
        except ImportError:
            return {"analogy": f"{a}:{b}::{c}:?", "predicted": "?", "candidates": []}

        candidates = []
        max_search = 20000
        for word_id in range(min(len(ID2WORD), max_search)):
            if word_id not in ID2WORD:
                continue
            word = ID2WORD[word_id]
            if len(word) < 2 or len(word) > 8:
                continue
            pure = re.sub(r'[\u064B-\u0652]', '', word)
            if not pure or len(pure) < 2:
                continue
            if pure in [a, b, c]:
                continue
            wv = _get_word_linear(pure)
            sim = float(np.dot(v_d_norm, wv) / (np.linalg.norm(wv) + 1e-10))
            candidates.append((float(sim), word))

        candidates.sort(reverse=True)
        top_k = candidates[:5]
        nearest_word = top_k[0][1] if top_k else "?"
        confidence = top_k[0][0] if top_k else 0.0

        return {
            "analogy": f"{a} : {b} :: {c} : ?",
            "method": "linear", "predicted": nearest_word,
            "confidence": round(confidence, 4),
            "delta_norm": round(float(np.linalg.norm(delta_v)), 4),
            "candidates": [(w, round(s, 4)) for s, w in top_k],
        }
    else:
        mv_a = _get_word_mv(a)
        mv_b = _get_word_mv(b)
        mv_c = _get_word_mv(c)
        delta = mv_b / mv_a
        mv_d = delta * mv_c
        nearest_word, nearest_sim = _find_nearest_word(mv_d)
        candidates = _find_top_k_words(mv_d, k=5)

        if nearest_word:
            mv_nearest = _get_word_mv(nearest_word)
            delta_back = mv_nearest / mv_c
            analogy_consistency = float(
                (delta.s * delta_back.s + np.dot(delta.v, delta_back.v) +
                 np.dot(delta.b, delta_back.b)) /
                (delta.norm() * delta_back.norm() + 1e-10))
        else:
            analogy_consistency = 0.0

        return {
            "analogy": f"{a} : {b} :: {c} : ?",
            "method": "geometric", "predicted": nearest_word,
            "confidence": round(abs(analogy_consistency), 4),
            "delta_operator": str(delta),
            "candidates": [(w, round(s, 4)) for w, s in candidates],
        }

def _find_nearest_word(mv: Multivector22) -> Tuple[str, float]:
    """البحث الهجين: ترشيح خطي سريع + مقارنة هندسية دقيقة."""
    try:
        from model.vocab import ID2WORD
    except ImportError:
        return ("?", 0.0)

    import re
    mv_norm = mv.norm()

    # مرحلة 1: ترشيح خطي
    linear_candidates = []
    max_linear = 60000
    for word_id in range(min(len(ID2WORD), max_linear)):
        if word_id not in ID2WORD:
            continue
        word = ID2WORD[word_id]
        if len(word) < 2 or len(word) > 8:
            continue
        pure = re.sub(r'[\u064B-\u0652]', '', word)
        if not pure or len(pure) < 2:
            continue
        wv = _get_word_linear(pure)
        sim = float(np.dot(mv.v, wv) / (mv_norm * np.linalg.norm(wv) + 1e-10))
        linear_candidates.append((sim, word))

    linear_candidates.sort(reverse=True)

    # مرحلة 2: مقارنة هندسية على أفضل 50
    best_word = "?"
    best_sim = -1.0
    for sim_l, word in linear_candidates[:50]:
        pure = re.sub(r'[\u064B-\u0652]', '', word)
        word_mv = _get_word_mv(pure)
        sim = (mv.s * word_mv.s + np.dot(mv.v, word_mv.v) +
               np.dot(mv.b, word_mv.b)) / (mv_norm * word_mv.norm() + 1e-10)
        if sim > best_sim:
            best_sim = sim
            best_word = word

    return (best_word, float(best_sim))


def _find_top_k_words(mv: Multivector22, k: int = 5) -> List[Tuple[str, float]]:
    """أقرب k كلمة (بحث هجين)."""
    try:
        from model.vocab import ID2WORD
    except ImportError:
        return []

    import re, heapq
    mv_norm = mv.norm()

    linear_candidates = []
    max_linear = 60000
    for word_id in range(min(len(ID2WORD), max_linear)):
        if word_id not in ID2WORD:
            continue
        word = ID2WORD[word_id]
        if len(word) < 2 or len(word) > 8:
            continue
        pure = re.sub(r'[\u064B-\u0652]', '', word)
        if not pure or len(pure) < 2:
            continue
        wv = _get_word_linear(pure)
        sim = float(np.dot(mv.v, wv) / (mv_norm * np.linalg.norm(wv) + 1e-10))
        linear_candidates.append((sim, word))

    linear_candidates.sort(reverse=True)

    heap = []
    for sim_l, word in linear_candidates[:50]:
        pure = re.sub(r'[\u064B-\u0652]', '', word)
        word_mv = _get_word_mv(pure)
        sim = (mv.s * word_mv.s + np.dot(mv.v, word_mv.v) +
               np.dot(mv.b, word_mv.b)) / (mv_norm * word_mv.norm() + 1e-10)
        heapq.heappush(heap, (float(sim), word))
        if len(heap) > k:
            heapq.heappop(heap)

    return [(w, s) for s, w in sorted(heap, reverse=True)]

def _get_rotation_matrix(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """حساب مصفوفة الدوران الجيوديسي في الفضاء 22D التي تدور u إلى v."""
    u_norm = np.linalg.norm(u)
    v_norm = np.linalg.norm(v)
    if u_norm < 1e-10 or v_norm < 1e-10:
        return np.eye(22)
    u_u = u / u_norm
    v_v = v / v_norm
    
    cos_t = np.dot(u_u, v_v)
    if cos_t > 0.99999:
        return np.eye(22)
    if cos_t < -0.99999:
        # انعكاس كامل 180 درجة في مستوى عشوائي متعامد مع u_u
        rng = np.random.default_rng(42)
        while True:
            rand_v = rng.standard_normal(22)
            rand_v -= np.dot(rand_v, u_u) * u_u
            norm = np.linalg.norm(rand_v)
            if norm > 1e-5:
                u_perp = rand_v / norm
                break
        return np.eye(22) - 2.0 * (np.outer(u_u, u_u) + np.outer(u_perp, u_perp))
        
    u_perp = v_v - cos_t * u_u
    u_perp_norm = np.linalg.norm(u_perp)
    if u_perp_norm < 1e-10:
        return np.eye(22)
    u_perp = u_perp / u_perp_norm
    
    sin_t = np.sqrt(max(0.0, 1.0 - cos_t**2))
    
    # مصفوفة الدوران الجيوديسي في مستوى (u, u_perp)
    R = (np.eye(22) 
         + sin_t * (np.outer(u_perp, u_u) - np.outer(u_u, u_perp))
         + (cos_t - 1.0) * (np.outer(u_u, u_u) + np.outer(u_perp, u_perp)))
    return R


def _get_word_vector(word: str, vector_type: str = "linear") -> np.ndarray:
    """استرجاع المتجه الدلالي المطلوب حسب نوعه (خطي أو جزء متبقي أو إسقاط)."""
    if vector_type == "linear":
        return _get_word_linear(word)
    elif vector_type in ("remnant", "projection"):
        return compute_compound_word_vector(word, method=vector_type)
    else:
        raise ValueError(f"Unknown vector type: {vector_type}")


def build_general_antinomy_operator(pairs: List[Tuple[str, str]], method: str = "linear", vector_type: str = "linear"):
    """
    بناء مؤثر التضاد العام من أزواج الأضداد المعروفة.
    
    الطرق المتاحة:
      - linear: Δ = mean(v(neg) - v(pos))
      - division: Δ = mean(v(neg) / v(pos)) (مع حماية التقسيم على الصفر)
      - clifford: Δ = mean(MV(neg) / MV(pos)) (يتجاهل vector_type لأنه يعمل على multivectors)
      - rotation: متوسط مصفوفات الدوران الجيوديسية المسقطة عبر SVD
    """
    if method == "clifford":
        sum_mv = Multivector22()
        for w_pos, w_neg in pairs:
            mv_pos = _get_word_mv(w_pos)
            mv_neg = _get_word_mv(w_neg)
            delta = mv_neg / mv_pos  # mv_pos.inverse() * mv_neg
            sum_mv = sum_mv + delta
        n = len(pairs)
        return Multivector22(sum_mv.s / n, sum_mv.v / n, sum_mv.b / n)
        
    if method == "linear":
        deltas = []
        for w_pos, w_neg in pairs:
            v_pos = _get_word_vector(w_pos, vector_type)
            v_neg = _get_word_vector(w_neg, vector_type)
            deltas.append(v_neg - v_pos)
        return np.mean(deltas, axis=0)
        
    elif method == "division":
        deltas = []
        for w_pos, w_neg in pairs:
            v_pos = _get_word_vector(w_pos, vector_type)
            v_neg = _get_word_vector(w_neg, vector_type)
            div = np.where(np.abs(v_pos) > 1e-5, v_neg / v_pos, 0.0)
            deltas.append(div)
        return np.mean(deltas, axis=0)
        
    elif method == "rotation":
        R_list = []
        for w_pos, w_neg in pairs:
            u = _get_word_vector(w_pos, vector_type)
            v = _get_word_vector(w_neg, vector_type)
            R_list.append(_get_rotation_matrix(u, v))
        R_avg = np.mean(R_list, axis=0)
        
        # تنظيف المصفوفة باستخدام SVD للحصول على مصفوفة دوران حقيقية متعامدة
        U, _, Vt = np.linalg.svd(R_avg)
        R_clean = U @ Vt
        if np.linalg.det(R_clean) < 0:
            U[:, -1] *= -1
            R_clean = U @ Vt
        return R_clean
    else:
        raise ValueError(f"Unknown antinomy method: {method}")


def apply_antinomy_operator(word: str, operator, method: str = "linear", vector_type: str = "linear"):
    """تطبيق مؤثر التضاد العام على كلمة لإنتاج تمثيل دلالي للمضاد."""
    if method == "clifford":
        mv = _get_word_mv(word)
        mv_new = mv * operator
        mv_new.normalize()
        return mv_new
        
    v = _get_word_vector(word, vector_type)
    if method == "linear":
        v_new = v + operator
    elif method == "division":
        v_new = v * operator
    elif method == "rotation":
        v_new = operator @ v
    else:
        raise ValueError(f"Unknown antinomy method: {method}")
        
    nrm = np.linalg.norm(v_new)
    return v_new / nrm if nrm > 1e-10 else v_new

def predict_antonym(word: str, operator, method: str = "linear", vector_type: str = "linear", top_k: int = 5) -> List[Tuple[str, float]]:
    """التنبؤ بضد كلمة معينة والبحث عن أقرب الكلمات من المعجم."""
    import re
    word_clean = re.sub(r'[\u064B-\u0652]', '', word)
    
    fast_results = []
    if word_clean in FAST_ANTONYMS:
        antonym = FAST_ANTONYMS[word_clean]
        fast_results.append((antonym, 1.0))
        if len(fast_results) >= top_k:
            return fast_results
            
    if method == "clifford":
        mv_new = apply_antinomy_operator(word, operator, method)
        candidates = _find_top_k_words(mv_new, k=top_k + 2)
        filtered = []
        seen = {word_clean}
        for w, sim in fast_results:
            w_clean = re.sub(r'[\u064B-\u0652]', '', w)
            seen.add(w_clean)
            filtered.append((w, sim))
            
        for w, sim in candidates:
            w_clean = re.sub(r'[\u064B-\u0652]', '', w)
            if w_clean in seen:
                continue
            seen.add(w_clean)
            filtered.append((w, sim))
        return filtered[:top_k]
    else:
        target_v = apply_antinomy_operator(word, operator, method, vector_type)
        target_v_norm = np.linalg.norm(target_v)
        if target_v_norm < 1e-10:
            return fast_results[:top_k]
        
        try:
            from model.vocab import ID2WORD
        except ImportError:
            return fast_results[:top_k]
        
        candidates = []
        max_search = 60000
        for word_id in range(min(len(ID2WORD), max_search)):
            if word_id not in ID2WORD:
                continue
            cand = ID2WORD[word_id]
            cand_clean = re.sub(r'[\u064B-\u0652]', '', cand)
            if not cand_clean or len(cand_clean) < 2 or cand_clean == word_clean:
                continue
            wv = _get_word_vector(cand_clean, vector_type)
            sim = float(np.dot(target_v, wv) / (target_v_norm * np.linalg.norm(wv) + 1e-10))
            candidates.append((cand, sim))
        
        candidates.sort(key=lambda x: x[1], reverse=True)
        
        filtered = []
        seen = {word_clean}
        for w, sim in fast_results:
            w_clean = re.sub(r'[\u064B-\u0652]', '', w)
            seen.add(w_clean)
            filtered.append((w, sim))
            
        for w, sim in candidates:
            w_clean = re.sub(r'[\u064B-\u0652]', '', w)
            if w_clean in seen:
                continue
            seen.add(w_clean)
            filtered.append((w, sim))
            if len(filtered) >= top_k:
                break
        return filtered[:top_k]


def compute_compound_word_vector(word: str, method: str = "projection") -> np.ndarray:
    """
    حساب متجه 22D يمثل الكلمة المركبة من خلال اندماجها الهندسي.
    
    الطرق المتاحة:
      - remnant: الجزء المتجهي المتبقي فقط mv.v مطبعاً
      - projection: المتجه المتبقي + انكماش ثنائي المتجه mv.v + contract(mv.b, mv.v) مطبعاً
    """
    mv = _get_word_mv(word)
    
    if method == "remnant":
        v = mv.v.copy()
    elif method == "projection":
        contracted = mv._contract_bv(mv.b, mv.v)
        v = mv.v + contracted
    else:
        raise ValueError(f"Unknown compound method: {method}")
        
    norm = np.linalg.norm(v)
    if norm < 1e-10:
        # Fallback for words where the vector component is 0 (e.g. 2-letter words)
        v = _get_word_linear(word)
        norm = np.linalg.norm(v)
        
    if norm > 1e-10:
        v = v / norm
    return v


def classify_word_field(word: str) -> str:
    """تصنيف الكلمة إلى أحد الحقول الدلالية الثلاثة الرئيسية بناءً على حروفها."""
    from src.physics.word_physics import _normalize_letters
    pure = _normalize_letters(word)
    
    epistemology_chars = set("علمجهلصدقكذبحقبطل")
    sensory_chars = set("نورظلمحبرديموت")
    state_chars = set("قوضعرفكبرصغ")
    
    counts = {
        "epistemology": sum(1 for ch in pure if ch in epistemology_chars),
        "sensory": sum(1 for ch in pure if ch in sensory_chars),
        "state": sum(1 for ch in pure if ch in state_chars)
    }
    
    dominant = max(counts, key=counts.get)
    if counts[dominant] == 0:
        return "epistemology"
    return dominant


def build_field_specific_antinomy_operators(pairs: List[Tuple[str, str]], method: str = "linear", vector_type: str = "linear") -> Dict[str, any]:
    """بناء مؤثرات تضاد متخصصة لكل حقل دلالي من أزواج الأضداد."""
    pairs_by_field = {
        "epistemology": [],
        "sensory": [],
        "state": []
    }
    
    for pos, neg in pairs:
        field_pos = classify_word_field(pos)
        pairs_by_field[field_pos].append((pos, neg))
        
    operators = {}
    for field, field_pairs in pairs_by_field.items():
        if field_pairs:
            operators[field] = build_general_antinomy_operator(field_pairs, method=method, vector_type=vector_type)
        else:
            operators[field] = build_general_antinomy_operator(pairs, method=method, vector_type=vector_type)
            
    return operators


def apply_field_specific_antinomy_operator(word: str, operators: Dict[str, any], method: str = "linear", vector_type: str = "linear"):
    """تطبيق مؤثر التضاد المتخصص بالحقل الدلالي للكلمة."""
    field = classify_word_field(word)
    operator = operators.get(field)
    return apply_antinomy_operator(word, operator, method=method, vector_type=vector_type)

def benchmark_analogies() -> Dict:
    """اختبار دقة القياس الدلالي على 8 أزواج أضداد."""
    results = []
    correct = 0
    total = len(KNOWN_ANALOGIES)

    for a, b, c, expected_d in KNOWN_ANALOGIES:
        result = solve_analogy(a, b, c, method="linear")
        predicted = result["predicted"]
        is_correct = (predicted == expected_d)
        if is_correct:
            correct += 1
        results.append({
            "analogy": f"{a}:{b} :: {c}:{expected_d}",
            "predicted": predicted, "expected": expected_d,
            "correct": is_correct, "confidence": result["confidence"],
        })

    return {
        "total_analogies": total, "correct": correct,
        "accuracy": round(correct / total, 4) if total > 0 else 0,
        "results": results,
    }


def build_selective_negation_operator(word: str, vector_type: str = "linear", num_dims: int = 5) -> np.ndarray:
    """بناء مؤثر النفي الاختياري للأبعاد المهيمنة."""
    v = _get_word_vector(word, vector_type)
    indices = np.argsort(np.abs(v))[::-1][:num_dims]
    return indices


def apply_selective_negation(word: str, operator: np.ndarray, vector_type: str = "linear") -> np.ndarray:
    """تطبيق نفي المتجه الاختياري للأبعاد المحددة."""
    v = _get_word_vector(word, vector_type).copy()
    for idx in operator:
        v[idx] *= -1.0
    norm = np.linalg.norm(v)
    return v / norm if norm > 1e-10 else v


def analyze_semantic_affinity(word1: str, word2: str) -> Dict:
    """قياس التقارب والترادف والفرقان الدلالي بين الكلمات."""
    v1_lin = _get_word_linear(word1)
    v2_lin = _get_word_linear(word2)
    sim_lin = float(np.dot(v1_lin, v2_lin) / (np.linalg.norm(v1_lin) * np.linalg.norm(v2_lin) + 1e-10))

    v1_rem = compute_compound_word_vector(word1, method="remnant")
    v2_rem = compute_compound_word_vector(word2, method="remnant")
    sim_rem = float(np.dot(v1_rem, v2_rem))

    v1_proj = compute_compound_word_vector(word1, method="projection")
    v2_proj = compute_compound_word_vector(word2, method="projection")
    sim_proj = float(np.dot(v1_proj, v2_proj))

    shared_essence = float(abs(sim_proj))

    if sim_proj > 0.4:
        affinity_class = "close_synonym"
    elif sim_proj > 0.15:
        affinity_class = "semantic_affinity"
    elif sim_proj < -0.25:
        affinity_class = "divergence_antonym"
    else:
        affinity_class = "nuanced_relation"

    return {
        "word1": word1,
        "word2": word2,
        "similarity": {
            "linear": round(sim_lin, 4),
            "remnant": round(sim_rem, 4),
            "projection": round(sim_proj, 4)
        },
        "shared_essence": round(shared_essence, 4),
        "affinity_class": affinity_class
    }
