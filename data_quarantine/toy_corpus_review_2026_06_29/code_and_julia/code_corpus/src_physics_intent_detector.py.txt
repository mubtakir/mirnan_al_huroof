class IntentDetector:
    """يكتشف قصد جملة المستخدم بإسقاطها على متجهات الأغراض الثابتة."""

    def __init__(self):
        self.intent_names = list(_INTENT_CANONICAL.keys())
        self._intent_pvs = {}
        self._build_intent_attractors()

    def _phrase_pv(self, phrase):
        """حساب متجه طوري لجملة (متوسط كلماتها)."""
        words = phrase.split()
        if not words:
            return None
        pvs = []
        for w in words:
            try:
                pv = compute_extended_phase_vector(w)
                pvs.append(pv)
            except Exception:
                continue
        if not pvs:
            return None
        avg = np.mean(pvs, axis=0)
        nrm = np.linalg.norm(avg)
        if nrm > 1e-10:
            avg = avg / nrm
        return avg

    def _build_intent_attractors(self):
        """بناء متجه جاذب لكل intent من متوسط عباراته الكانونية."""
        for intent_name, phrases in _INTENT_CANONICAL.items():
            pvs = []
            for phrase in phrases:
                pv = self._phrase_pv(phrase)
                if pv is not None:
                    pvs.append(pv)
            if pvs:
                attractor = np.mean(pvs, axis=0)
                nrm = np.linalg.norm(attractor)
                if nrm > 1e-10:
                    attractor = attractor / nrm
                self._intent_pvs[intent_name] = attractor

    def detect(self, text):
        """كشف القصد عبر الطور أولاً، مع تعزيز بالكلمات المفتاحية.

        الفلسفة: القصد ليس كلمة — بل موجة. متجه الطور للجملة يُسقط
        على متجهات الأقاصد الكانونية. الكلمات المفتاحية تُعزّز الثقة
        لكنها لا تُحدد القصد وحدها.

        Returns dict:
            intent: القصد المكتشف
            confidence: 0.0–1.0
            distances: مسافة لكل intent
        """
        if not text:
            return {"intent": "STATEMENT", "confidence": 0.0, "distances": {}}

        # ١) الإسقاط الطوري — الأساس
        input_pv = self._phrase_pv(text)
        if input_pv is None:
            return {"intent": "STATEMENT", "confidence": 0.0, "distances": {}}

        distances = {}
        for name, attractor in self._intent_pvs.items():
            cos_sim = float(np.mean(np.cos(input_pv[:INTENT_DIM] - attractor[:INTENT_DIM])))
            distances[name] = cos_sim

        # ٢) تعزيز الكلمات المفتاحية (يعزز الثقة، لا يحدد القصد)
        text_lower = text.lower()
        words_set = set(text_lower.split())
        keyword_boost = {}
        for intent_name, kws in _INTENT_KEYWORDS.items():
            for kw in sorted(kws, key=len, reverse=True):
                multi = kw.split()
                if len(multi) > 1:
                    if kw in text_lower:
                        keyword_boost[intent_name] = keyword_boost.get(intent_name, 0) + 1
                        break
                else:
                    if kw in words_set:
                        keyword_boost[intent_name] = keyword_boost.get(intent_name, 0) + 1
                        break

        # تعزيز المسافة الطورية للقصد المطابق كلماتياً
        for intent_name, boost in keyword_boost.items():
            if intent_name in distances:
                distances[intent_name] += boost * 0.15  # تعزيز خفيف

        best = max(distances, key=distances.get)
        raw_confidence = distances[best]
        min_dist = min(distances.values())
        confidence = max(0.0, min(1.0, (raw_confidence - min_dist + 0.3)))

        return {
            "intent": best,
            "confidence": confidence,
            "distances": distances,
        }

        # 2) المتجهات الطورية
        input_pv = self._phrase_pv(text)
        if input_pv is None:
            return {"intent": "STATEMENT", "confidence": 0.0, "distances": {}}

        distances = {}
        for name, attractor in self._intent_pvs.items():
            cos_sim = float(np.mean(np.cos(input_pv[:INTENT_DIM] - attractor[:INTENT_DIM])))
            distances[name] = cos_sim

        best = max(distances, key=distances.get)
        confidence = max(0.0, (distances[best] - min(distances.values())))

        return {
            "intent": best,
            "confidence": round(min(confidence * 5.0, 1.0), 4),
            "distances": {k: round(v, 4) for k, v in distances.items()},
        }

    def get_intent_pv(self, intent_name):
        return self._intent_pvs.get(intent_name)

    def get_default_response_target(self, intent_name):
        """إرجاع متجه طوري هدف للرد بناءً على القصد.

        لكل intent، نأخذ آخر كلمة في عباراته الكانونية كهدف للرد.
        """
        phrases = _INTENT_CANONICAL.get(intent_name, [])
        if not phrases:
            return None
        # نأخذ آخر كلمة من أول عبارة كانتقدير سريع
        last_words = []
        for p in phrases[:3]:
            ws = p.split()
            if ws:
                try:
                    pv = compute_extended_phase_vector(ws[-1])
                    last_words.append(pv)
                except Exception:
                    continue
        if not last_words:
            return None
        target = np.mean(last_words, axis=0)
        nrm = np.linalg.norm(target)
        if nrm > 1e-10:
            target = target / nrm
        return target

    def intent_label_ar(self, intent_name):
        labels = {
            "GREETING": "تحية",
            "QUESTION": "سؤال",
            "COMMAND": "أمر",
            "REQUEST": "طلب",
            "FAREWELL": "وداع",
            "STATEMENT": "خبر",
            "OPINION": "رأي",
            "SUGGESTION": "اقتراح",
            "COMPLAINT": "شكوى",
            "PROMISE": "وعد",
            "THANK": "شكر",
        }
        return labels.get(intent_name, intent_name)

    @property
    def intents(self):
        return list(self._intent_pvs.keys())
