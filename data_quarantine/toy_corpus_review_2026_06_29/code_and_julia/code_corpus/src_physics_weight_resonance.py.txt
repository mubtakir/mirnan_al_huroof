class WeightResonanceEngine:
    def __init__(self, morpho_engine=None):
        self.morpho = morpho_engine
        self._weight_pv_cache = {}
        for name, vec in _WEIGHT_EMBEDDINGS.items():
            nrm = np.linalg.norm(vec)
            if nrm > 1e-10:
                self._weight_pv_cache[name] = vec / nrm
            else:
                self._weight_pv_cache[name] = vec

    def get_weight(self, word):
        if self.morpho is None:
            return None
        analysis = self.morpho.analyze(word)
        if not analysis['has_morph']:
            return None
        return analysis['weight']

    def get_weight_pv(self, weight_name):
        return self._weight_pv_cache.get(weight_name, np.zeros(PHASE_DIM))

    def resonance(self, word1_weight, word2_weight):
        if word1_weight is None or word2_weight is None:
            return 0.0
        if word1_weight not in _WEIGHT_EMBEDDINGS or word2_weight not in _WEIGHT_EMBEDDINGS:
            return 0.0
        pv1 = self._weight_pv_cache[word1_weight]
        pv2 = self._weight_pv_cache[word2_weight]
        return float(np.mean(np.cos(pv1 - pv2)))

    def transition_score(self, prev_weight, word_weight):
        if prev_weight is None or word_weight is None:
            return 0.0
        trans = _WEIGHT_TRANSITIONS.get(prev_weight, {})
        return trans.get(word_weight, 0.0)

    def weight_density(self, words):
        weight_counts = {}
        for w in words:
            weight = self.get_weight(w)
            if weight:
                weight_counts[weight] = weight_counts.get(weight, 0) + 1
        if not weight_counts:
            return 0.0
        max_count = max(weight_counts.values())
        return max_count / max(len(words), 1)
