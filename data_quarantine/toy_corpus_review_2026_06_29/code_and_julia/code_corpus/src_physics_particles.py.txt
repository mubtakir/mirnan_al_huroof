_ALL_PARTICLES: Set[str] = (
    _HURUF_AL_JARR
    | _HURUF_AL_NASB
    | _HURUF_AL_JAZM
    | _HURUF_AL_ATF
    | _ADAWAT_AL_SHART
    | _ADAWAT_AL_NAFY
    | _ADAWAT_AL_ISTIFHAM
    | _COMPOUND_PARTICLES
)

# ─── دوال الاستعلام ───
_PARTICLE_CACHE: dict = {}


def is_particle(word: str) -> bool:
    if not word or len(word) <= 1:
        return False
    if word in _PARTICLE_CACHE:
        return _PARTICLE_CACHE[word]
    result = word in _ALL_PARTICLES
    _PARTICLE_CACHE[word] = result
    return result


def get_particle_role(word: str) -> Optional[str]:
    if word in _HURUF_AL_JARR:
        return "JARR"
    if word in _HURUF_AL_NASB:
        return "NASB"
    if word in _HURUF_AL_JAZM:
        # تفريق بين الجزم والنفي والشرط
        if word in ('لَمْ', 'لَمَّا', 'لم', 'لما'):
            return "NAFY_JAZM"
        if word in _ADAWAT_AL_SHART:
            return "SHART_JAZM"
        return "JAZM"
    if word in _HURUF_AL_ATF:
        return "ATF"
    if word in _ADAWAT_AL_SHART:
        return "SHART"
    if word in _ADAWAT_AL_NAFY:
        if word in ('لَمْ', 'لَمَّا', 'لَم', 'لمَّا'):
            return "NAFY_JAZM"
        if word in ('لَنْ', 'لن'):
            return "NAFY_NASB"
        return "NAFY"
    if word in _ADAWAT_AL_ISTIFHAM:
        return "ISTIFHAM"
    if word in _COMPOUND_PARTICLES:
        return "COMPOUND"
    # حروف النصب المتبقية
    if word in ('كَيْ', 'كي', 'لِكَيْ', 'لكي', 'إِذَنْ', 'إذن', 'أَنْ', 'أنْ'):
        return "NASB"
    return None


def clear_cache():
    _PARTICLE_CACHE.clear()
