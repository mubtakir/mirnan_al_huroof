def get_category(word):
    """إرجاع قائمة الفئات التي تنتمي إليها الكلمة."""
    results = []
    for cat, words in CATEGORIES.items():
        if word in words:
            results.append(cat)
    return results


def category_bonus(word, context_words):
    """مكافأة تصنيف دلالي.

    إذا كانت الكلمة من نفس فئة كلمات السياق → مكافأة.
    إذا كانت من فئة مختلفة → لا مكافأة.
    """
    word_cats = get_category(word)
    if not word_cats:
        return 0.0

    for cw in context_words[-3:]:
        cw_cats = get_category(cw)
        if cw_cats:
            overlap = set(word_cats) & set(cw_cats)
            if overlap:
                return 0.3  # مكافأة ثابتة
    return 0.0


def context_category(context_words):
    """تحديد الفئة الدلالية للسياق بناءً على كلماته."""
    cat_counts = {}
    for w in context_words:
        cats = get_category(w)
        for c in cats:
            cat_counts[c] = cat_counts.get(c, 0) + 1
    if cat_counts:
        return max(cat_counts, key=cat_counts.get)
    return None


def is_same_category(word1, word2):
    """التحقق مما إذا كانت كلمتان تنتميان لنفس الفئة الدلالية."""
    cats1 = set(get_category(word1))
    cats2 = set(get_category(word2))
    return bool(cats1 & cats2)
