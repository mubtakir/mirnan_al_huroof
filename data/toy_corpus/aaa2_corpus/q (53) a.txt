== 🎭 أبيات شعرية فيها طباق وتضاد

= من الشعر العربي القديم:

=أبو الطيب المتنبي:=
> أَزُورُهُم وَسَوادُ اللَيلِ يَشفَعُ لي
> وَأَنثَني وَبَياضُ الصُبحِ يُغري بي

(ليل / صبح – سواد / بياض – يشفع / يغري)

=المتنبي أيضاً:=
> وَما الحَياةُ وَنَفسي بَعدَما عَرَفَت
> وَلا المَماتُ لِما بي عِندَهُ خَبَرُ

(حياة / ممات)

=أبو فراس الحمداني:=
> أَرى المَوتَ لِلأَغنياءِ اِختِيارا
> وَلِلفُقَراءِ الأَعِزّاءِ حَتمُ

(أغنياء / فقراء – اختيار / حتم)

=زهير بن أبي سلمى:=
> وَمَن يَجعَلِ المَعروفَ مِن دونِ عِرضِهِ
> يَفِرهُ وَمَن لا يَتَّقِ الشَتمَ يُشتَمِ

(معروف / شتم – يفره / يشتم)

=عنترة بن شداد:=
> وَلَقَد ذَكَرتُكِ وَالرِماحُ نَواهِلٌ
> مِنّي وَبيضُ الهِندِ تَقطُرُ مِن دَمي

(نواهل / تقطر – ترطيب مقابل إدماء)

=أبو العتاهية:=
> أَلا إِنَّما الدُنيا غُروبٌ وَمَطلَعُ
> وَعَيشُ الفَتى فيها عَذابٌ وَمَطمَعُ

(غروب / مطلع – عذاب / مطمع)

=بشار بن برد:=
> إِذا كُنتَ في الأَحياءِ تَبكي صَبابَةً
> فَأَنتَ عَلى الأَمواتِ أَجدَرُ أَن تَبكي

(أحياء / أموات)

=أبو نواس:=
> وَما الذَيلُ إِلّا لِلكِرامِ وَإِنَّما
> يُجَرُّ عَلى الهاماتِ ذَيلُ المَذَلَّةِ

(كرام / مذلة)

=المتنبي:=
> إِذا الجودُ لَم يُرزَق خَلاصاً مِنَ الأَذى
> فَلا الحَمدُ مَكسوباً وَلا المالُ باقِيا

(جود / أذى – حمد / مال)

=

= من الشعر الحديث والمعاصر:

=أحمد شوقي:=
> وَالحُرُّ يَصدُقُ حينَ يَنطِقُ أَو يَفي
> وَالعَبدُ يَكذِبُ في الوَفاءِ وَيَمطُلُ

(حر / عبد – صدق / كذب – وفاء / مطل)

=حافظ إبراهيم:=
> الأُمُّ مَدرَسَةٌ إِذا أَعدَدتَها
> أَعدَدتَ شَعباً طَيِّبَ الأَعراقِ

(أعددت / أعددت – تضاد سياقي بين الإهمال والإعداد)

=محمود درويش:=
> وَلَنا في الحَياةِ مَآتِمُنا
> وَلَنا في المَماتِ مَوَالِدُنا

(حياة / ممات – مآتم / مواليد)

=إيليا أبو ماضي:=
> أَنا لا أَقولُ إِذا المَواسِمُ أَقبَلَت
> هَذا رَبيعٌ بَل أَقولُ خَريفُ

(ربيع / خريف)

=أبو القاسم الشابي:=
> إِذا الشَّعبُ يَوماً أَرادَ الحَياةَ
> فَلا بُدَّ أَن يَستَجيبَ القَدَر
> وَلا بُدَّ لِلَّيلِ أَن يَنجَلي
> وَلا بُدَّ لِلقَيدِ أَن يَنكَسِر

(ليل / ينجلي – قيد / ينكسر)

=نازك الملائكة:=
> في قَلبِنا صَمتٌ طَويلٌ حالِكٌ
> وَعَلى شِفاهِ الصَمتِ أَلفُ سُؤالِ

(صمت / سؤال)

=السياب:=
> عَيناكِ لَيلٌ وَصُبحٌ
> تَأتي وَتَمضي كَطَيفِ

(ليل / صبح – تأتي / تمضي)

=

= أبيات مرتجلة على نفس النسق (محاكاة):

ليلي بوجهكِ مشرقٌ متنوِّرُ
يا من بنور جبينه أتكوّرُ
فالغصن إن مال النسيم بروضه
فالقلب إن خطر الخيال تحيَّرا
(ليل / نور – مال / تحيّر)

قيظ الفراق وهمسه متنفَّسُ
برد الوصال بطيفه متلمَّسُ
إن غاب شخصك فالضلوع توجّع
إن جئت فالنبض السعيد مقدَّس
(قيظ / برد – غاب / جئت)

أبكيتَني فأثرتَ شجويَ و الأسى
أضحكتني فأعدتَ نبضيَ للورى
أظمأتَ روحي في حضورك والهوى
أرويتَ قلبي حين غبتَ عن الكرى
(أبكيت / أضحكت – أظمأت / أرويت – حضور / غياب)

قم للوصال فقد جفانيَ هاجري
فالوصل عندي جنة المتعبدِ
إن أقبلت أيام أُنسيَ بالمنى
فالعمر في الهجران شوكٌ في يدي
(وصل / هجر – أقبلت / هجران – جنة / شوك)

=

هل تريد أن ننتقل إلى صيغة أخرى، كالأمثال الشعبية أو الحِكم القصيرة التي تحوي تضاداً؟

تفضّل، إليك مجموعة واسعة من =الأمثال الشعبية= و=الحِكم القصيرة= التي بُني كثير منها على التضاد والطباق بين الكلمات:

=

== 🏛️ أمثال شعبية فيها تضاد

= أمثال عربية فصيحة (قديمة):

ما طار طير وارتفع إلا كما طار وقع.

أعطِ القوس باريها.

إن الغنى والعز في القناعة، والذل في الطمع.

خير الكلام ما قلّ ودلّ.

ربّ كلمة قالت لصاحبها: دعني.

الصيفَ ضيّعتِ اللبن.

قبل الرماء تُملأ الكنائن.

الساكت عن الحق شيطان أخرس.

إن شرّ الناس من باع دينه بدنيا غيره.

خير الناس من نفع الناس.

الكذب داء والصدق دواء.

يد واحدة لا تصفّق.

الجار قبل الدار.

الرفيق قبل الطريق.

خبي قرشك الأبيض ليومك الأسود.

=

== 📜 حِكم قصيرة مبنية على التضاد

= حكم عربية قديمة:

مَن ساسَ الناسَ باللينِ دانوا له، ومَن ساسَهم بالخُشونةِ خانوه.

العلم في الصِّغَر كالنقش في الحجر.

خير الأصحاب مَن يدلّك على الخير.

ليس العاقل مَن يعرف الخير من الشر، ولكن مَن يعرف خير الشرّين.

مَن تأنّى نالَ ما تمنّى.

القناعة كنز لا يفنى.

آفة النفس الشَّرَه، وزينة النفس الوَرَع.

خير الغنى القنوع، وشرّ الفقر الخضوع.

مَن طلب العلا سهر الليالي.

الرفق بالضعيف من شيم الشريف.

لسان العاقل وراء قلبه، وقلب الأحمق وراء لسانه.

عجلة الأب في الندامة، وعجلة الأم في السلامة.

العلم نور والجهل ظلام.

الصبر مفتاح الفرج.

دوام الحال من المحال.

=

= حكم مترجمة إلى العربية (عالمية):

أعطني خبزاً أمسك بيدك، أعطني عملاً أمسك بقلبك.

الحياة إما مغامرة جريئة أو لا شيء.

الأفعال أبلغ من الأقوال.

فكّر في النهاية قبل البداية.

ليس كل ما يلمع ذهباً.

مَن يزرع الريح يحصد العاصفة.

في الاتحاد قوة، وفي الفرقة ضعف.

خير صديق هو الكتاب، وشر عدو هو الجهل.

لا تؤجّل عمل اليوم إلى الغد.

الحكمة أن تعرف متى تتكلم ومتى تصمت.

=

= حِكم مختصرة بكلمتين متضادتين (شعارات):

صمت يتكلم.

غياب حاضر.

موت في حياة.

ظلام منير.

فراق دائم الوصال.

حزن يفرح.

ضعف قوة.

قيد حرية.

أمل في يأس.

سكون عاصف.

=

= حكم مرتجلة على النسق ذاته:

اصبر على مرّ الجفاء لعلّه... يأتي بحلو الوصل بعد نفارِ
فالدهر ذو وجهين: باسط كفّه... طوراً، وقابضها على الأقدارِ

قم لليل المعتم الحالكِ، يتبعه صبح منير ضاحكُ
لا تيأسنّ فإن بعد عُسرِكَ... يُسراً يُنير دروبك السالكة

إن الضحى يمحو ظلام دجنّة... والغصن يُورق بعد طول جفاف
فاصبر فإن الصبر مفتاح الرضا... والهم ينجاب انجياب ضعاف

كونوا كشمسٍ تنشر النورَ للورى... لا كالظلام يضمّهم في غيهب
فالعلم ريّ للنفوس إذا ظمئت... والجهل سمٌ للعقول فجنّبوا

=

هل تحب أن أنتقل بعدها إلى =الألغاز= أو =الطرائف اللغوية= المبنية على الأضداد؟

تفضّل، إليك مجموعة منتقاة =بالفصحى فقط= من الألغاز اللغوية والطرائف الأدبية المبنية على التضاد والطباق:

== 🧩 ألغاز لغوية مبنية على التضاد

= ألغاز شعرية:

=اللغز الأول:=
> أنا شيءٌ عجيبٌ في صفاتي
> أُرى في العين أحياناً وأمّحي
> فإن أبصرتني أبصرت ضدّي
> وإن تغمض فإني لا أبرحُ

(الجواب: =الظل=، فهو يظهر في الضوء ويختفي في الظلام، وفيه معنى التضاد بين الإبصار والإمحاء)

=

=اللغز الثاني:=
> أنا ضدّي ونفسي في آنِ
> أُحبّ وأبغض بلا توانِ
> فإن غبتُ كنتُ أشدَّ حضوراً
> وإن حضرتُ كنتُ بلا كيانِ

(الجواب: =الصمت=، فهو ضد الكلام، وحضوره غياب للصوت، وغيابه حضور للنطق)

=

=اللغز الثالث:=
> وما أنا؟ إن نطقتُ صمتُّ
> وإن صمتُّ تكلمّ الناسُ فيّ
> أُحلّي تارةً وأمرُّ أخرى
> فمن لي بالذي ينجو إليّ؟

(الجواب: =السرّ=، إن أُفشي صمتّ بانكشافه، وإن صمتّ عنه تكلم الناس به، يحلو بالإخفاء ويمرّ بالإذاعة)

=

=اللغز الرابع:=
> اسمي وضدّي في حروف ثلاثة
> إن زال أوّلي غدا ضدّي بباقيه
> فما أنا؟

(الجواب: =نوم= – بحذف النون يصير "وم" قريباً من اليقظة لا لفظاً، ولكن من أمثلة الألغاز: كلمة "نام" ضدها "صحا")

=

=اللغز الخامس:=
> أنا ذو لونين: بياضٌ وسوادُ
> ولي في كل يومٍ ميعادُ
> إذا ما زرتكم عمّ رقادٌ
> وإن ولّيتُ أيقظكم سهادُ

(الجواب: =الليل والنهار= معاً، فالإقبال نوم والإدبار يقظة)

=

=اللغز السادس:=
> أنا شيء إذا أَقبَلَ المرءُ نحوي
> جعلتُ القريبَ منه بعيدا
> وإن أَدبَرَ المرءُ عنّي قليلاً
> رأى البعيدَ من خلفي قريبا

(الجواب: =المرآة=، تقرّب البعيد حين يُدبر وتُبعد القريب حين يُقبل)

=

=اللغز السابع:=
> من أيّ شيءٍ خُلقتُ؟ لستُ أدري
> غير أني أُحِبُّ وأكرَهُ معا
> أجمع الماء والنارَ بكفّي
> وأُذيب الثلجَ والصخرَ الأصمّا

(الجواب: =اللسان= أو =الكلمة=، فيهما جمع الأضداد من الحب والكره، والإصلاح والإفساد)

=

=اللغز الثامن:=
> أنا طويلٌ ولستُ بالطويلْ
> وأنا قصيرٌ ولستُ بالقصيرْ
> ليس لي ظلّ إذا مشيتُ
> وأنا كلُّ الظلال حين أسيرْ

(الجواب: =النهار والليل= – نهار طويل بلا ظل، وليل قصير لكنه يملأ الدنيا ظلالاً)

=

=اللغز التاسع:=
> إذا جئتكم كنتُ في غاية القرب
> وإن غبتُ عنكم فأنا في غاية البعد
> فبالوصل أُنسى وبالهجر أُذكر
> فمن يعرف اسمي؟

(الجواب: =الغائب الحاضر=، كالموت والحياة في الذاكرة، أو =الشوق= الذي يزيد بالبعد وينقص بالقرب)

=

=اللغز العاشر:=
> أنا الضعيف الذي يغلب الأقوياء
> وأنا الصامت الذي يُفصح بالدواء
> أدخل البيوت بلا استئذان
> وأُخرج أهلها بلا عدوان

(الجواب: =المرض=، ضعيف في ذاته يغلب الأقوياء، صامت لكنه يُفصح عن الداء)

=

== 🎭 طرائف لغوية وأدبية مبنية على التضاد

= الطرفة الأولى:

قيل لأعرابي: صِف لنا الخمر. قال:  
="هي حلوةٌ فُوَّهة، مُرَّةٌ عُقباهُا. لذيذةٌ شُرْبَتُها، وخيمةٌ صحبتُها. تمنح الفقيرَ غِنى الكِبر، وتُعطي الغنيَّ فقرَ الذلّ. تبعث النشاط في العظام، وتُورث الخمول في الأحلام."=  
فقيل: هذا ضدّ في وصفك! قال: وكذلك هي في الطبع.

=

= الطرفة الثانية:

سأل نحويٌّ تلميذه: ما ضدّ "قام"؟  
قال: "قعد".

قال: وما ضدّ "نام"؟  
قال: "صحا".

قال: وما ضدّ "الليل"؟  
قال: "النهار".

قال: فما ضدّ "الضدّ" إذن؟  
فبهت التلميذ، ثم قال: ="الوفاق"=، فضحك النحوي وقال: بل هو ="المِثْل"=.

=

= الطرفة الثالثة:

وقف أعرابي على باب والٍ، فأذن له، فلما دخل قال الوالي: سل حاجتك.

قال الأعرابي:  
="أيها الأمير، أنا رجلٌ غنيٌّ فقير، قريبٌ بعيد، عزيزٌ ذليل. أمّا غناي ففي قناعتي، وأما فقري ففي طمع غيري إليّ. وأما قربي ففي طاعتك، وأما بعدي ففي معصيتك. وأما عزّي ففي أخلاقي، وأما ذلّي ففي حاجتي إليك."=  
فأعجب الوالي ببلاغته وقضى حاجته.

=

= الطرفة الرابعة:

كتب أديب إلى صديق له:  
="أما بعد، فإني أكتب إليك من مكانٍ بعيدِ القرب، قريبِ البعد، طويلِ المسافة، قصيرِ الشوق. فإن غبتَ عني فأنت أقرب إليّ من حبل الوريد، وإن حضرتَ فغيابك عن قلبي أبعد من الثريّا."=  
فكتب إليه الصديق: ="فهمتُ أنك مشتاقٌ جائع، فتعال نأكل ونجوع معاً!"=

=

= الطرفة الخامسة:

دخل رجل على الحجّاج بن يوسف وكان جواداً شجاعاً ولكنه جَبانٌ بخيل في الحرب والمال.

فقال له: ="أيها الأمير، إن فيك خصالاً متضادات: جوداً وبخلاً، إقداماً وجُبناً، ليناً وعنفاً!"=  
فقال الحجاج: ="صدقت؛ فجوادي لمن يستحق، وبخلي على من لا يستحق. وإقدامي على العدو، وجبني عن الظلم. وليني للضعيف، وعنفي للمعاند."=

=

= الطرفة السادسة:

تخاصم زوجان أمام قاضٍ أعرابي، فقال الزوج:  
="هي نائمةٌ يقظى، ساكتةٌ نطوق، غائبةٌ حاضرة!"=  
فقال القاضي: فسّر كلامك.

قال: ="أما نومها فغفلتها عن حقي، وأما يقظتها فانتباهها لحقها. وأما سكوتها فإذا طلبتُ، وأما نطقها فإذا طلبتْ. وأما غيابها فإذا حلّ الضيق، وأما حضورها فإذا جاء الوسع."=  
فقالت المرأة: ="صدق، وأزيدك: هو قويٌّ ضعيف، غنيٌّ فقير!"=  
قال: فسّري.

قالت: ="قوي على أهله، ضعيف عن واجبه. غني بحب نفسه، فقير بحب غيره!"=

=

= الطرفة السابعة:

مدح شاعر أميراً بقوله:  
> =طلعتَ علينا طلعةَ الشمس في الضحى=  
> =وأشرقتَ كالبدر المنير بلا ظلامْ=

فقال أحد الحاضرين: الشمس في الضحى والبدر المنير لا يكونان معاً!  
فأجابه الشاعر: ="بل هما معاً في وجه الأمير، فإنه يجمع بين عزّ النهار ووقار الليل."=

=

= الطرفة الثامنة:

كتب أعرابي إلى ابنه يوصيه:  
="يا بني: كن غنياً في فقرك، عزيزاً في ذلّك، حيّاً في موتك. أما الغنى في الفقر فالقناعة، وأما العز في الذل فحسن الخلق، وأما الحياة في الموت فالأثر الطيب."=

=

= الطرفة التاسعة:

سأل أحدهم حكيماً: كيف حالك؟  
قال:  
="أنا في عافيةٍ سقيمة، ونعمةٍ نقمة. عافيتي أني مؤمن، وسقمي أني عاصٍ. ونعمتي أني أعرف ربي، ونقمتي أني لا أعمل بما أعرف."=

=

= الطرفة العاشرة:

قال بعض البلغاء:  
="الدنيا وجهان في مرآة واحدة: إقبالها إدبار، وإدبارها إقبال. وغناها فقر، وفقرها غنى. فخذ من إقبالها عبرة، ومن إدبارها عظة."=

=

هل ننتقل بعدها إلى =أقوال مأثورة عن الحكماء والفلاسفة= مبنية على الطباق، أم تريد مجالاً آخر في الفصحى؟

تفضّل مجموعة من =أقوال الحكماء والفلاسفة والأدباء= بالفصحى، مبنية على الطباق والتضاد:

=

== 📜 أقوال الحكماء والفلاسفة

= من التراث العربي الإسلامي:

=الإمام علي بن أبي طالب (رضي الله عنه):=
> "الناس أعداء ما جهلوا، وأولياء ما عرفوا."
(أعداء / أولياء – جهلوا / عرفوا)

> "إذا أقبلت الدنيا على أحد أعارته محاسن غيره، وإذا أدبرت عنه سلبته محاسن نفسه."
(أقبلت / أدبرت – أعارت / سلبت)

> "الغنى في الغربة وطن، والفقر في الوطن غربة."
(غنى / فقر – وطن / غربة)

> "خير الناس من نفع الناس، وشر الناس من ضرّ الناس."
(خير / شر – نفع / ضرّ)

> "ربّ قول أشدّ من صول."
(قول / صول)

> "العلم ينادي العمل، فإن أجابه وإلا ارتحل."
(علم / عمل – أجاب / ارتحل)

> "ما أضمر أحد شيئاً إلا ظهر في فلتات لسانه وصفحات وجهه."
(أضمر / ظهر)

=

=الإمام الشافعي (رضي الله عنه):=
> "الحرّ من راعى وداد لحظة، وانتمى لمن أفاده لفظة."
(حر / عبد – ضمني)

> "العلم ما نفع، ليس العلم ما حُفظ."
(نفع / حفظ)

> "من تكلّف ما لا يعرف ضيّع ما يعرف."
(تكلّف / عرف – ضيّع / يعرف)

=

=ابن المقفّع:=
> "الغنيّ من استغنى بالله، والفقير من افتقر إلى الناس."
(غني / فقير – استغنى / افتقر)

> "رأس الحكمة مخافة الله، ورأس الجهل الأمن من مكره."
(حكمة / جهل – مخافة / أمن)

> "خير الكلام ما قلّ ودلّ، ولم يطل فيملّ."
(قلّ / طال – دلّ / ملّ)

=

=الجاحظ:=
> "الكتاب نعم الجليس، إن صمتَ ألهى، وإن نطقتَ أنصت، وإن ضحكتَ تبسّم، وإن بكيتَ شارك."
(صمت / نطق – ألهى / أنصت – ضحك / بكاء)

> "العاقل من جعل لسانه تابعاً لعقله، والأحمق من جعل عقله تابعاً للسانه."
(عاقل / أحمق – لسان / عقل)

=

=ابن حزم الأندلسي:=
> "من طلب الآخرة أدرك الدنيا، ومن طلب الدنيا فاتته الآخرة."
(آخرة / دنيا – أدرك / فات)

> "العزّ في الطاعة، والذلّ في المعصية."
(عز / ذل – طاعة / معصية)

=

=ابن عطاء الله السكندري:=
> "ربّ معصية أورثت ذلاً وانكساراً خير من طاعة أورثت عزاً واستكباراً."
(معصية / طاعة – ذل / عز – انكسار / استكبار)

> "اجتهادك فيما ضُمِن لك، وتقصيرك فيما طُلب منك، دليل على طمس البصيرة."
(اجتهاد / تقصير – ضمن / طلب)

=

= من أقوال الزهاد والعبّاد:

=الحسن البصري:=
> "ما زالت التقوى بالمتقين حتى تركوا كثيراً من الحلال مخافة الحرام."
(حلال / حرام)

> "المرء في الدنيا على جناح سفر: يحلّ ويرتحل، ويقبل ويدبر."
(يحل / يرتحل – يقبل / يدبر)

=

=مالك بن دينار:=
> "من تباعد من زهرة الدنيا فذلك الغني، ومن تقارب منها فذلك الفقير."
(تباعد / تقارب – غني / فقير)

=

=الفضيل بن عياض:=
> "من عرف الناس استراح، ومن جهل الناس تعب."
(عرف / جهل – استراح / تعب)

> "بئس القوم لا يرونك أهلاً إلا لما عندك، ونعم القوم لا يرونك أهلاً إلا لما فيك."
(بئس / نعم – عندك / فيك)

=

=أبو سليمان الداراني:=
> "ربما أدخل الظلمة على العبد من خير يزجيه إليه أكثر مما يدخل عليه من شر يكفّه عنه."
(خير / شر – يزجيه / يكفه)

=

= من أقوال الحكماء غير العرب (مترجمة إلى الفصحى):

=سقراط:=
> "الحكمة أن تعرف أنك لا تعرف."
(تعرف / لا تعرف)

> "اللسان الطويل دليل اليد القصيرة."
(طويل / قصيرة)

> "العلم الحقيقي أن تفرق بين ما تدري وما لا تدري."
(تدري / لا تدري)

=

=أفلاطون:=
> "أعقل الناس من يعرف نقصه، وأحمق الناس من يجهل عيبه."
(أعقل / أحمق – يعرف / يجهل – نقص / عيب)

> "حبّ الظهور يكسر الظهور."
(ظهور / ظهور – جناس وطباق ضمني)

=

=أرسطو:=
> "الجاهل يؤكّد، والعالم يشكّ، والعاقل يتروّى."
(جاهل / عاقل – يؤكد / يتروى)

> "المتوسط في كل شيء هو الفضيلة، والطرفان رذيلة."
(وسط / طرفان – فضيلة / رذيلة)

=

=كونفوشيوس:=
> "من غلب نفسه كان أقوى ممن غلب مدينة."
(غلب نفسه / غلب مدينة)

> "الصمت صديق الحكمة، والكلام عدوّ الطيش."
(صمت / كلام – صديق / عدو – حكمة / طيش)

=

=لأوتسو (الحكيم الصيني):=
> "اللين يغلب الصلب، والضعيف يتغلّب على القوي."
(لين / صلب – ضعيف / قوي)

> "من عرف الناس صمت، ومن عرف نفسه نطق."
(عرف / جهل – صمت / نطق)

=

=الفيلسوف الهندي بوديدهارما:=
> "الجهل أصل كل شر، والمعرفة أصل كل خير."
(جهل / معرفة – شر / خير)

> "من لا يملك شيئاً يملك كل شيء."
(لا يملك / يملك – شيئاً / كل شيء)

=

= من روائع الأدباء المتأخرين:

=مصطفى صادق الرافعي:=
> "السكوت حكمة لا يفهمها الثرثار، كما أن الكلام فن لا يتقنه الأخرس."
(سكوت / كلام – ثرثار / أخرس)

(شتاء / ربيع – ليل / فجر)

=

=عباس محمود العقاد:=
> "الفرق بين العاقل والأحمق: العاقل يطلب الحقيقة، والأحمق يطلب التصديق."
(عاقل / أحمق – حقيقة / تصديق)

> "الصبر نصف الإيمان، واليأس نصف الكفر."
(صبر / يأس – إيمان / كفر)

=

=ميخائيل نعيمة:=
> "الغني من كثرت حسناته، لا من كثرت أمواله."
(غني / فقير – حسنات / أموال)

> "كم من صامت أبلغ من متكلم، وكم من ساكن أسرع من متحرك."
(صامت / متكلم – ساكن / متحرك)

=

=جبران خليل جبران:=
> "أنتم أحياء ما دمتم في ذواتكم، وأموات ما دمتم في غيركم."
(أحياء / أموات – ذواتكم / غيركم)

> "الفرح المفرط حزن، والحزن العميق فرح خفي."
(فرح / حزن – مفرط / عميق)

=

=علي الطنطاوي:=
> "العلم بلا عمل جنون، والعمل بلا علم كفر."
(علم / عمل – جنون / كفر)

> "الحرية في القيد إذا كان القيد من ذاتك، والقيد في الحرية إذا كانت الحرية من شهوتك."
(حرية / قيد – ذات / شهوة)

=

=محمود محمد شاكر:=
> "الضعيف من أظهر قوته، والقوي من أخفى ضعفه."
(ضعيف / قوي – أظهر / أخفى)

= أقوال مأثورة قصيرة جامعة:

العز في التواضع، والذل في التكبر.

الصبر مفتاح الفرج، واليأس مفتاح الحزن.

الغنى غنى النفس، والفقر فقر النفس.

خير الناس أحسنهم خُلقاً، وشرهم أسوؤهم خُلقاً.

الدنيا سجن المؤمن وجنة الكافر.

الجود من الموجود، والبخل من المعدوم.

من عرف الحق هانت عليه الأباطيل.

مفتاح الشر الغضب، ومفتاح الخير الحلم.

العاقل من اتعظ بغيره، والأحمق من اتعظ بنفسه.

من زاد في حب الدنيا نقص من حب الآخرة.

الليل ستر العيوب، والنهار كاشفها.

الكلمة الطيبة صدقة، والكلمة الخبيثة فتنة.

== 🎙️ خُطب ووصايا مبنية على التضاد

= الخطبة الأولى: خطبة قس بن ساعدة الإيادي (المشهورة)

> "أيها الناس، اسمعوا وعوا، وإذا وعيتم فانتفعوا. إنه =من عاش مات، ومن مات فات، وكل ما هو آتٍ آت=. ليل داج، ونهار ساج، وسماء ذات أبراج، ونجوم تزهر، وبحار تزخر، وجبال مرساة، وأرض مدحاة. ما لي أرى الناس =يذهبون فلا يرجعون=؟ أرضوا بالمقام فأقاموا، أم تُركوا هُناك فناموا؟"

(عاش / مات – يذهبون / يرجعون – أقاموا / ناموا – ليل / نهار)

=

= الخطبة الثانية: خطبة للإمام علي (رضي الله عنه) في وصف الدنيا

> "أما بعد، فإن الدنيا دار =مَن لا دار له=، ولها يجمع =مَن لا عقل له=، وبها يفرح =مَن لا معرفة له=. فمن أبصرها =أبصرته عبراً=، ومن أبصرته =أعمته غرراً=. =حلوها مرّ، وصفوها كدر=، وسرورها حزن، وعزّها ذلّ. =يُقبِل منها ما يُدبِر عنك، ويُدبِر عنك ما يُقبِل منها=. فأفضل الناس فيها أعرفهم بها، وأزهدهم فيها، وأرغبهم عنها."

(أبصر / أعمى – حلو / مر – صفو / كدر – سرور / حزن – عز / ذل – يقبل / يدبر)

=

= الخطبة الثالثة: خطبة لعمر بن عبد العزيز (رضي الله عنه)

> "أيها الناس، إن الدنيا دار =مُنية لا بقاء=، و=عزّ لا دوام=، و=غنى لا تمام=. =من أقبلت عليه فتنته=، و=من أدبرت عنه أحزنته=. إنما مثلها كمثل =الظل الزائل=، و=الربيع الحائل=. فخذوا من =صحّتكم لمرضكم=، ومن =شبابكم لهرمكم=، ومن =حياتكم لموتكم=، ومن =غناكم لفقركم=."

(إقبال / إدبار – صحة / مرض – شباب / هرم – حياة / موت – غنى / فقر)

=

= الخطبة الرابعة: خطبة للحسن البصري (رحمه الله)

> "يا ابن آدم، =إنك في نقصان منذ يوم ولدت=، فأنت =إلى الموت تمضي، وفي القبر تُودَع، وإلى البعث تُحشَر=. =كم من مستقبل يوماً لا يستكمله! وكم من مؤمِّل غداً لا يدركه!= إنك لو رأيت =الأموات أحياءً= لما غرّتك الحياة. ولو أبصرت =الأحياء أمواتاً= لعرفت حقيقة الفناء. =فاعمل لدار البقاء في دار الفناء=."

(نقصان / تمام – موت / بعث – أموات / أحياء – بقاء / فناء)

=

= الخطبة الخامسة: خطبة لأبي بكر الصديق (رضي الله عنه)

> "أما بعد، فإني قد وُلِّيت عليكم =ولست بخيركم=، فإن =أحسنت فأعينوني=، وإن =أسأت فقوّموني=. =الصدق أمانة، والكذب خيانة=. =الضعيف فيكم قوي عندي حتى آخذ الحق له=، و=القوي فيكم ضعيف عندي حتى آخذ الحق منه=. أطيعوني =ما أطعت الله=، فإذا =عصيت الله فلا طاعة لي عليكم=."

(أحسن / أساء – صدق / كذب – ضعيف / قوي – طاعة / معصية)

=

= الخطبة السادسة: خطبة لزياد بن أبيه (الخطبة البتراء)

> "أما بعد، فإن =الجهل بالحق لا يدفع باطلاً=، و=العلم بالباطل لا يجلب حقاً=. وإن =الغيّ لا يقاد بالهدى=، و=الرشد لا يردّ بالضلال=. ألا وإن =الطاعة عصمة، والمعصية ندامة=. =فمن أطاعنا أكرمناه=، و=من عصانا عاقبناه=. =اللين منا رحمة، والشدة منا نقمة=."

(جهل / علم – حق / باطل – غي / رشد – طاعة / معصية – لين / شدة)

=

= الوصية الأولى: وصية أعرابي لابنه

> "يا بني، =إن الدنيا بحر عميق، والناس فيه بين غريق وناج=. =فمن سبح سبح، ومن غرق غرق=. فكن =في الغنى متواضعاً، وفي الفقر متعففاً=. =اضحك مع الضاحكين قليلاً، وابكِ مع الباكين كثيراً=. =أعطِ من سألك عفواً، وامنع من منعك صفحاً=. =صِل من قطعك، وأعطِ من حرمك=."

(غريق / ناج – غنى / فقر – ضحك / بكاء – أعطِ / امنع – صِل / قطع)

=

= الوصية الثانية: وصية حكيم لتلميذه

> "اعلم أن =العلم شجرة، أصلها مرة وثمرها حلو=. و=الجهل شجرة، أصلها حلو وثمرها مر=. فاصبر على مرارة البداية تدرك حلاوة النهاية. وكن =في الغنى شاكراً، وفي الفقر صابراً=. =واخفض جناحك لمن ولاك، وارفع رأسك لمن عاداك=. =تكلم بالحق ولو كان مُرّاً، ودع الباطل ولو كان حُلواً=."

(علم / جهل – مر / حلو – غنى / فقر – خفض / رفع – حق / باطل)

=

= الوصية الثالثة: وصية أم لابنتها ليلة زفافها

> "يا بنية، إنك =تغادرين بيتاً صغيراً إلى بيت كبير=، و=تنتقلين من عزّ أبيك إلى عزّ زوجك=. فكوني =في لينك قوية، وفي قوتك لينة=. =في حضورك غائبة عن عيوبه، وفي غيابك حاضرة بحفظ سره=. =قولي له خيراً تسمعي خيراً=. =فإن أعطى فاشكري، وإن منع فاصبري=. =واجعلي سمعك له طاعة، وبصرك له متعة=."

(صغير / كبير – لين / قوة – حضور / غياب – أعطى / منع)

=

= الوصية الرابعة: وصية قائد لجنوده

> "يا جنود الحق، =لا تَضعُفوا أمام القوي، ولا تَقْوَوا على الضعيف=. =كونوا في السلام حماماً، وفي الحرب صقوراً=. =أطعموا الجائع، واسقوا الظامئ=. =ارحموا الأسير، واضربوا الغادر=. =أكرموا من سلّم، واهزموا من هجم=. =وابنوا ما هُدم، وأصلحوا ما فسد=."

(ضعيف / قوي – سلام / حرب – أطعموا / اسقوا – ارحموا / اضربوا – سلم / هجم – هدم / أصلح)

=

= الوصية الخامسة: وصية تاجر لشركائه

> "يا معشر التجار، =الربح في الصدق، والخسارة في الكذب=. =والغنى في الأمانة، والفقر في الخيانة=. =والرواج في البذل، والكساد في الشحّ=. =فكونوا في البيع أسخياء، وفي الشراء أوفياء=. =من أربحتموه أخلص لكم، ومن خسرتموه تبرأ منكم=. =خذوا من الغني قليلاً تعيشوا، وأعطوا الفقير كثيراً تُحمدوا=."

(ربح / خسارة – صدق / كذب – غنى / فقر – رواج / كساد – بيع / شراء – غني / فقير)

=

== 🏟️ مناظرات أدبية مبنية على التضاد

= المناظرة الأولى: مناظرة بين الليل والنهار

قال =الليل=:  
> "أنا ستر العيوب، وكنز الأسرار. =فيّ يسكن الخائفون، ويرتاح المتعبون=. =بي تُقضى الحاجات، وتُنال البركات=. أنا =زمن العبادة والتهجد=، ووقت =الخلوة والتفرد=."

فأجاب =النهار=:  
> "أنا كاشف الحقائق، ومجلي البصائر. =فيّ يسعى العاملون، ويجتهد الطالبون=. =بي تُدرك الأرزاق، وتُعمر الآفاق=. أنا =زمن الكسب والجهاد=، ووقت =العطاء والامتداد=."

فحكم بينهما =الغسق=:  
> "كلاكما جميل، =فأنتما توأمان لا يفترقان=. =بظلامك يُعرف النور، وبنورك يُعرف الظلام=. =فسبحان من جعلكما آيتين: الليل آية للسكون، والنهار آية للحركة=."

=

= المناظرة الثانية: مناظرة بين العقل والقلب

قال =العقل=:  
> "أنا ميزان العدل، ومرآة الصواب. =بي يُفرّق بين الحق والباطل=، و=يُميّز بين النافع والضار=. أنا =أساس العلم، وقاعدة الحكمة=. =من أطاعني رشد، ومن عصاني غوى=."

فأجاب =القلب=:  
> "أنا منبع الحب، وينبوع الرحمة. =بي تتحرك المشاعر، وتسمو النفوس=. أنا =أساس الإيمان، وقاعدة الإخلاص=. =من أطاعني عشق، ومن عصاني قسا=. =العقل بارد وأنا دافئ، وهو يُبعد وأنا أُقرّب=."

فحكم بينهما =اللسان=:  
> "أنتما جناحا الإنسان، =إن رجح العقل طار إلى المعرفة=، و=إن رجح القلب حلق إلى المحبة=. =فالعقل بلا قلب جفاف، والقلب بلا عقل ضياع=. فسبحان من جمعكما في صدر واحد."

=

= المناظرة الثالثة: مناظرة بين السيف والقلم

قال =السيف=:  
> "أنا قاطع الهامات، وهادم الممالك. =بي يُنصر المظلوم، ويُقهر الظالم=. أنا =لسان القوة، وحجة الجبابرة=. =من ملكني ملك، ومن فقدني هلك=."

فأجاب =القلم=:  
> "أنا صانع الأمم، وباني الحضارات. =بي يُنشر العلم، ويُقام العدل=. أنا =لسان الحكمة، وحجة العقلاء=. =من ملكني خُلّد، ومن فقدني جُهِل=. =أنت تهدم وأنا أبني، أنت تُميت وأنا أُحيي=."

فحكم بينهما =الميزان=:  
> "أنتما أخوان في خدمة الحق. =فالسيف بلا قلم طغيان، والقلم بلا سيف خذلان=. =بالسيف يُحمى القلم، وبالقلم يُوجَّه السيف=. فلا غنى لأحدكما عن الآخر."

=

= المناظرة الرابعة: مناظرة بين الصمت والكلام

قال =الكلام=:  
> "أنا وسيلة البيان، وطريق التعبير. =بي يُعرف المجهول، ويُشرح المعقول=. أنا =دليل العقل، ومرآة الفكر=. =من أحسنني كُرِّم، ومن أخطأني نُدِم=."

فأجاب =الصمت=:  
> "أنا كنز الحكمة، وستر الغيوب. =بي تُصان الأسرار، وتُحفظ الوقار=. أنا =دليل العقل، ومأوى الحلم=. =من أطالني سلم، ومن قصّرني ندم=. =الكلام يجرّ إلى الزلل، وأنا أحمي من الخطل=."

فحكم بينهما =العاقل=:  
> "لكل مقام مقال. =ففي العلم كلام، وفي الحلم صمت=. =خير الكلام ما قلّ ودلّ، وخير الصمت ما طال وطاب=. =فالصمت في موضع الكلام عجز، والكلام في موضع الصمت جهل=."

=

= المناظرة الخامسة: مناظرة بين الشباب والشيب

قال =الشباب=:  
> "أنا ربيع الحياة، وزينة الوجود. =بي تقوى الأبدان، وتعلو الهمم=. أنا =زمن القوة والنشاط=، ووقت =العزم والانطلاق=. =من عاشني عاش في سرور، ومن فقدني مات في قبور=."

فأجاب =الشيب=:  
> "أنا تاج الوقار، ودليل الحكمة. =بي تكمل التجارب، وتنضج العقول=. أنا =زمن الرزانة والاتزان=، ووقت =العبرة والامتحان=. =من عاشني عاش في اعتبار، ومن عابني عاش في اغترار=. =أنت قوة الجسد، وأنا قوة الروح=."

فحكم بينهما =الدهر=:  
> "كلاكما طوران من أطوار العمر. =الشباب بداية والشيب نهاية=. =فمن أحسن في شبابه كُرِّم في شيبه=. =فخذوا من الشباب جدّاً، ومن الشيب حدّاً=. =فرب شاب في حكمة شيخ، ورب شيخ في طيش شاب=."

=

= المناظرة السادسة: مناظرة بين الدنيا والآخرة

قالت =الدنيا=:  
> "أنا دار المتاع، ومنزل السعة. =بي تُدرك اللذات، وتُنال الشهوات=. أنا =عاجلة حاضرة=، و=نقد معجّل=. =من أقبل عليّ استغنى، ومن أدبر عني افتقر=."

فأجابت =الآخرة=:  
> "أنا دار القرار، ومنزل الخلود. =بي تُدرك النعيم، وتُنال الرضوان=. أنا =آجلة باقية=، و=وعد مؤجّل=. =من عمل لي ربح، ومن أعرض عني خسر=. =أنتِ فانية وأنا باقية، أنتِ مزرعة وأنا مثمرة=."

فحكم بينهما =العبد العاقل=:  
> "صدقتما. =فالدنيا مزرعة الآخرة=. =من زرع فيها خيراً حصد في الآخرة خيراً=، و=من زرع شراً حصد ندامة=. =فخذوا من الدنيا للآخرة، ومن الفناء للبقاء=. =فما الدنيا إلا جسر، فلا تعمروها وإنما اعبروها=."

=

هل تريد الانتقال إلى =الأشعار التعليمية (كألفية ابن مالك ونحوها)= المبنية على التضاد، أم إلى =أمثلة من الرسائل الأدبية= (كالمقامات)؟

تفضّل، إليك =الأشعار التعليمية= (مثل ألفية ابن مالك وشروحها) وما يشبهها من منظومات فيها تضاد، ثم =الرسائل الأدبية والمقامات= المبنية على الطباق، وكلّها بالفصحى:

=

== 📘 الأشعار التعليمية والمنظومات المبنية على التضاد

= من ألفية ابن مالك في النحو (مع شواهد الطباق):

=في المعرفة والنكرة:=
> وَ=المَعْرِفَةُ= اسْمٌ قَد تَعَيَّنَ مُطْلَقَا
> =وَنَكِرَةٌ= شَاعَت فَكُنْ مُحَقِّقَا

(معرفة / نكرة – تعين / شاع)

=

=في المذكر والمؤنث:=
> =مُذَكَّرٌ= يَخلُو مِنَ العَلامَه
> =وَمُؤَنَّثٌ= تَلحَقُهُ العَلامَه

(مذكر / مؤنث – يخلو / تلحقه)

=

=في الإعراب والبناء:=
> =مُعرَبٌ= وَالآخِرُ =مَبنيٌّ= عُرِف
> وَالاسمُ مِنهُ =مُعربٌ= وَمُنصَرِف
> وَ=غَيرُ مُنصَرِفٍ= بِمَا قَد وُصِف

(معرب / مبني – منصرف / غير منصرف)

=

=في الجامد والمشتق:=
> =جَامِدٌ= مَا لَم يَكُن مِن فِعلِهِ
> =وَمُشتَقٌّ= مَا كَانَ ذَا تَنَقُّلِه

(جامد / مشتق)

=

=في الفعل اللازم والمتعدي:=
> =لَازِمٌ= مَا لَم يَصِل لِغَيرِهِ
> =مُتَعَدٍّ= مَا وَصَلَ الأَمرَ إِلَيه

(لازم / متعدٍ – لم يصل / وصل)

=

=في المبني للمعلوم والمبني للمجهول:=
> =مَعلُومُهُ= فَاعِلُهُ مَذكُورُ
> =وَمَجهُولُهُ= عَن ذِكرِهِ مَستُورُ

(معلوم / مجهول – مذكور / مستور)

=

= من ألفية ابن مالك في الصرف:

=في الصحيح والمعتل:=
> =صَحِيحُهُ= مَا خَلَت أُصُولُه
> مِن حُرفِ عِلَّهٍ =وَمُعتَلٌّ= لَه

(صحيح / معتل)

=

=في المجرد والمزيد:=
> =مُجَرَّدٌ= مَا زَادَ حَرفُهُ عَلَى
> ثَلاثَةٍ =وَمُزَّيَدٌ= مَا قَد حَلا

(مجرد / مزيد)

=

=في التام والناقص:=
> =تَامٌ= وَ=نَاقِصٌ= بِحَذفِ الآخِرِ
> يَأتي كَذَا =مَمدُودُهُ= وَ=المَقصُورُ=

(تام / ناقص – ممدود / مقصور)

=

= من لامية الأفعال لابن مالك:

> =فَالمَاضِيَ= افتَح آخِرَاً وَ=المُضَارِعَا=
> فَارفَع وَ=أَمَّا الأَمرُ= فَاجزِم مُطلَقَا
> =مَبنِيُّهُ= مَا أَشبَهَ الحَرفَ =وَمُع=
> =رَبٌّ= سِوَاهُ فَاحفَظَنَّهُ تَنتَفِع

(ماض / مضارع / أمر – مبني / معرب)

=

= من تحفة الأطفال في التجويد (الجمزوري):

> =والمَدُّ= أَصلِيٌّ =وَفَرعِيٌّ= لَهُ
> وَسَمِّ أَولاً طَبِيعِياً وَهُو مَا
> لَا تَوقُّفٌ لَهُ عَلَى سَبَبْ
> وَلَا بِدُونِهِ الحُرُوفُ تُكتَسَبْ
> بَل أَيُّ حَرفٍ غَيرُ هَمزٍ أَو سُكُون
> جَا بَعدَ مَدٍّ فَهوَ =طَبعِيٌّ= يَكُون
> وَ=الآخَرُ الفَرعِيُّ= مَوقُوفٌ عَلَى
> سَبَبْ كَهَمزٍ أَو سُكُونٍ مُسجَلَا

(طبيعي / فرعي – أصل / فرع)

=

= من متن الآجرومية نظماً (للعمريطي):

> =وَاسمٌ وَفِعلٌ ثُمَّ حَرفٌ= تُدرِي
> أَقسَامُ كَلَامِ العَرَبِ استَقَرَّت
> =فَالاسمُ= بِالخَفضِ وَبِالتَّنوِينِ
> وَدُخُولِ الأَلِفِ وَاللَّامِ يُبِينُ

(اسم / فعل / حرف)

=

= منظومة في الأضداد (منسوبة لبعض النحاة):

> وَ=ضِدُّ قَامَ= =قَعَدَ= اتِّزَانَا
> وَ=ضِدُّ نَامَ= =صَحَا= كَمَا بَانَا
> =وَاللَّيلُ ضِدُّ النَّهَارِ= حَقَّا
> =وَالشَّرقُ ضِدُّ الغَربِ= صَدَقَا
> =وَالضَّوؤُ ضِدُّ الظَّلَامِ= حَتما
> =وَالحَرُّ ضِدُّ البَردِ= حُكما
> =وَالخَيرُ ضِدُّ الشَّرِّ= تَبيَانَا
> =وَالصِّدقُ ضِدُّ الكِذبِ= بُرهَانَا
> =وَالنَّفعُ ضِدُّ الضُّرِّ= تَقرِيرَا
> =وَالسَّهلُ ضِدُّ الصَّعبِ= تَحرِيرَا

=

= منظومة في الطباق (تعليمية):

> الطِّبَاقُ فِي النَّظمِ هُوَ الجَمعُ بَينَ لَفظَينِ
> =ضِدَّينِ= فَمَيز عَنهُمَا غَيرَ الضِّدَّينِ
> =فَالحُسنُ ضِدُّ القُبحِ= فِي تَبيَانِ
> =وَالطُّولُ ضِدُّ القِصَرِ= افتَرَقَا
> =وَالغِنَى ضِدُّ الفَقرِ= وَ=العِزُّ= ارتَفَع
> =ضِدُّ الذُّلِ= وَ=النُّورُ ضِدُّ الظُّلمَةِ=
> =وَالضَّحِكُ ضِدُّ البُكَاءِ= فِي الشِّدَّةِ
> =وَالشَّجَاعَةُ ضِدُّ الجُبنِ= فِي الفِكرَةِ

=

= أبيات في الأضداد من التراث (لأبي الطيب اللغوي):

> وَفِي كَلَامِ العَرَبِ =الأَضْدَادُ=
> تَجِيءُ =مِنهَا الوَاحِدُ الفَرَّادُ=
> =فَالصَّرِيمُ= لِليَلِ وَ=الصَّرِيمُ=
> لِلصُّبحِ هَذَا عِندَهُم قَدِيمُ
> =وَالظَّنُّ= لِليَقِينِ وَ=الظَّنُّ= لِلشَّك
> =وَالرَّجَبُ= التَّعظِيمُ وَ=الرَّجَبُ= الفَك
> =وَالسَّدَفُ= الظَّلَامُ وَ=السَّدَفُ= الضَّو
> ءُ كَذَاكَ =الجَونُ= لِلحُلوِ وَلِلمُر

=

== 📜 الرسائل الأدبية والمقامات المبنية على التضاد

= من مقامات بديع الزمان الهمذاني:

=المقامة الحلوانية (في وصف حاله):=
> "أَنَا الْيَوْمَ =بَيْنَ غِنًى وَفَقْرٍ=، وَ=بَيْنَ صِحَّةٍ وَسُقْمٍ=، وَ=بَيْنَ عِزٍّ وَذُلٍّ=. =فَغِنَايَ بِالْيَأْسِ، وَفَقْرِي بِالْحِرْصِ=. وَ=صِحَّتِي بِالْقَنَاعَةِ، وَسُقْمِي بِالطَّمَعِ=. وَ=عِزِّي بِالتَّقَلُّلِ، وَذُلِّي بِالتَّكَثُّرِ=. =فَمَنْ رَضِيَ بِالْقَلِيلِ عَزَّ، وَمَنْ طَلَبَ الْكَثِيرَ ذَلَّ=."

(غنى / فقر – صحة / سقم – عز / ذل – قليل / كثير)

=

=المقامة البغدادية (في وصف طالب العلم):=
> "رَأَيْتُهُ =يَدْخُلُ وَيَخْرُجُ=، =يَجْلِسُ وَيَقُومُ=، =يَضْحَكُ وَيَبْكِي=. فَقُلْتُ: مَا شَأْنُكَ؟ قَالَ: =الْعِلْمُ أَحْيَانِي وَأَمَاتَنِي=. =فَأَنَا بَيْنَ يَقِينٍ وَشَكٍّ=، وَ=بَيْنَ فَهْمٍ وَوَهْمٍ=. =أَعْرِفُ فَأَجْهَلُ، وَأَجْهَلُ فَأَعْرِفُ=. قُلْتُ: وَكَيْفَ ذَلِكَ؟ قَالَ: =كُلَّمَا عَلِمْتُ شَيْئاً جَهِلْتُ أَشْيَاءَ=."

(يدخل / يخرج – يجلس / يقوم – يضحك / يبكي – أحيى / أمات – يقين / شك – أعرف / أجهل)

=

=المقامة الرصافية (في وصف الفتى المتلون):=
> "فَتًى =إنْ شِئْتَ وَجَدْتَهُ فِي الْحَرْبِ أَسَداً، وَفِي السِّلْمِ حَمَلاً=. =وَإِنْ شِئْتَ رَأَيْتَهُ فِي الْغِنَى مُتَوَاضِعاً، وَفِي الْفَقْرِ مُتَرَفِّعاً=. =يَصْمُتُ فَيُفْهِمُ، وَيَنْطِقُ فَيُلْهِمُ=. =يُقْبِلُ عَلَى الْجَاهِلِ فَيُعَلِّمُهُ، وَيُدْبِرُ عَنِ الْعَالِمِ فَيَتَعَلَّمُ مِنْهُ=."

(حرب / سلم – غنى / فقر – يصمت / ينطق – جاهل / عالم – يقبل / يدبر)

=

= من رسائل الجاحظ (رسالة التربيع والتدوير):

> "اعْلَمْ أَنَّ =النَّاسَ فِي الدُّنْيَا بَيْنَ مَغْبُونٍ وَمَحْسُودٍ، وَبَيْنَ ظَالِمٍ وَمَظْلُومٍ، وَبَيْنَ غَنِيٍّ وَفَقِيرٍ، وَبَيْنَ عَالِمٍ وَجَاهِلٍ=. فَلَا تَجِدُ =غَنِيّاً إلَّا وَهُوَ مَحْسُودٌ، وَلَا فَقِيراً إلَّا وَهُوَ مَغْبُونٌ=. فَإِنِ اسْتَطَعْتَ أَنْ تَكُونَ =غَنِيَّ النَّفْسِ فِي فَقْرِ الْيَدِ= فَافْعَلْ."

(مغبون / محسود – ظالم / مظلوم – غني / فقير – عالم / جاهل)

=

= من رسائل ابن العميد (إلى الصاحب بن عباد):

> "كَتَبْتُ إلَيْكَ =مِنْ بُعْدٍ هُوَ قُرْبٌ، وَمِنْ غَيْبَةٍ هِيَ حُضُورٌ=. فَإِنَّ =الْأَحِبَّةَ إذَا تَبَاعَدَتْ أَبْدَانُهُمْ تَقَارَبَتْ أَرْوَاحُهُمْ=، و=إذَا غَابَتْ وُجُوهُهُمْ حَضَرَتْ مَوَدَّاتُهُمْ=. وَلَقَدْ صِرْتُ =فِي بُعْدِكَ كَالْجِسْمِ بِلَا رُوحٍ، وَفِي قُرْبِكَ كَالرُّوحِ بِلَا جِسْمٍ=. =فَزُرْنِي تَجِدْنِي قَرِيباً، وَاهْجُرْنِي تَجِدْنِي غَرِيباً=."

(بعد / قرب – غيبة / حضور – تباعدت / تقاربت – قريباً / غريباً)

=

= من رسائل أبي العلاء المعري (رسالة الغفران - بعض نثرها):

> "قَالَ الشَّيْخُ: =الْحَيَاةُ مَوْتٌ، وَالْمَوْتُ حَيَاةٌ=. =فَمَا الْعَيْشُ إلَّا كَالْمَنَامِ، وَمَا الْمَمَاتُ إلَّا كَالْيَقَظَةِ=. =يَضْحَكُ الْحَيُّ وَهُوَ إلَى الْمَوْتِ سَائِرٌ، وَيَبْكِي الْمَيْتُ وَهُوَ إلَى الْحَيَاةِ صَائِرٌ=. =فَسُبْحَانَ مَنْ أَضْحَكَ وَأَبْكَى، وَأَحْيَا وَأَمَاتَ=."

(حياة / موت – عيش / منام – يضحك / يبكي – أحيا / أمات)

=

= من مقامات الحريري:

=المقامة الدينارية (في وصف الكرم والبخل):=
> "فَمِنْهُمْ =مَنْ يُعْطِي وَيُمْسِكُ، وَيَبْذُلُ وَيَبْخَلُ=، =يُوسِعُ فِي مَوْضِعِ التَّوْسِعَةِ، وَيَقْتُرُ فِي مَوْطِنِ التَّقْتِيرِ=. فَذَاكَ =الْكَرِيمُ الْبَخِيلُ=، وَ=الْجَوَادُ الضَّنِينُ=. وَأَمَّا =الْبَخِيلُ الْحَقُّ فَمَنْ يُمْسِكُ عَن الْوَاجِبِ، وَيَمْنَعُ الْحَقَّ اللَّازِبَ=."

(يعطي / يمسك – يبذل / يبخل – يوسع / يقتر – كريم / بخيل)

=

=المقامة الصنعانية (في وصف القوة والضعف):=
> "وَرَأَيْتُ =الْقَوِيَّ ضَعِيفاً، وَالضَّعِيفَ قَوِيّاً=. =فَالْقَوِيُّ مَنْ قَوِيَ عَلَى نَفْسِهِ، وَالضَّعِيفُ مَنْ ضَعُفَ عَنْ شَهْوَتِهِ=. =وَالْغَنِيُّ مَنْ غَنِيَ بِرَبِّهِ، وَالْفَقِيرُ مَنْ فَقِيرٌ إلَى خَلْقِهِ=."

(قوي / ضعيف – غني / فقير – بربه / خلقه)

=

= من رسالة ابن زيدون إلى ابن جهور:

> "=أَقْبَلَتْ عَلَيَّ الدُّنْيَا بَعْدَ إدْبَارٍ، وَأَشْرَقَتْ لِي بَعْدَ إظْلَامٍ، وَأَنْبَسَطَتْ لِي بَعْدَ انْقِبَاضٍ=. فَكُنْتُ =كَالْغَائِبِ حَضَرَ، وَالنَّائِمِ اسْتَيْقَظَ=. =فَالْحَمْدُ لِلَّهِ الَّذِي جَعَلَ بَعْدَ الْعُسْرِ يُسْراً، وَبَعْدَ الضِّيقِ فَرَجاً=."

(أقبلت / إدبار – أشرقت / إظلام – انبسطت / انقباض – غائب / حضر – عسر / يسر – ضيق / فرج)

=

= من رسالة ولي الدين اليمني (في العتاب):

> "كِتَابِي إلَيْكَ =يَحْمِلُ عِتَاباً هُوَ عِشْقٌ، وَهَجْراً هُوَ وَصْلٌ، وَبُعْداً هُوَ قُرْبٌ=. فَإِنِّي =أَشْكُوكَ إلَيْكَ، وَأَفِرُّ مِنْكَ إلَيْكَ=. =فَأنْتَ قَلْبِي وَهَلْ يَشْكُو الْقَلْبُ إلَّا إلَى الْقَلْبِ؟="

(عتاب / عشق – هجر / وصل – بعد / قرب – أشكوك / إليك – أفر / إليك)

=

= من الرسائل الإخوانية للصابي:

> "أَخِي: =إنْ أَنَا أَغْضَبْتُكَ فَقَدْ أَرْضَيْتُ ضَمِيرِي=. =وَإِنْ أَنَا أَبْعَدْتُكَ فَقَدْ قَرَّبْتُ صِدْقِي=. =فَخُذْ مِنِّي الْمُرَّ تَنَلِ الْحُلْوَ، وَاقْبَلِ الْهَجْرَ تَجِدِ الْوَصْلَ=. =فَرُبَّ عِتَابٍ جَلَبَ رِضًى، وَرُبَّ فِرَاقٍ أَدَّى إلَى لِقَاءٍ=."

(أغضبت / أرضيت – أبعدت / قربت – مر / حلو – هجر / وصل – عتاب / رضا – فراق / لقاء)

=

== 📖 نماذج من المقامات والرسائل بأسلوب "بين... و..."

> "أَنَا بَيْنَ =الْحُبِّ وَالْبُغْضِ=، وَبَيْنَ =الْأَمَلِ وَالْيَأْسِ=، وَبَيْنَ =الصَّبْرِ وَالْجَزَعِ=. =فَحُبِّي يَدْعُونِي إلَى الْوِصَالِ، وَبُغْضِي يَدْفَعُنِي إلَى الْفِرَاقِ=. =وَأَمَلِي يَعِدُنِي بِاللِّقَاءِ، وَيَأْسِي يُنْذِرُنِي بِالْفَوَاتِ=. =فَيَا لَيْتَنِي أَدْرِي: أَأَنَا فِي غِنًى أَمْ فَقْرٍ؟ وَفِي قُرْبٍ أَمْ بُعْدٍ؟="

=

> "إلَى مَوْلَايَ الْأَجَلِّ: =إنْ كَانَ صَمْتِي عَنْكَ رِضًى فَأَنَا رَاضٍ، وَإِنْ كَانَ كَلَامِي إلَيْكَ سَخَطاً فَأَنَا سَاخِطٌ=. =فَدَاوِنِي بِالْقُرْبِ أُشْفَ، وَاقْتُلْنِي بِالْبُعْدِ أَمُتْ=. =فَمَا أَنَا فِي هَذَا الزَّمَانِ إلَّا بَيْنَ حَيَاةٍ وَمَمَاتٍ=."

=

> "=فِي قَلْبِي نَارٌ وَفِي عَيْنِي مَاءٌ=. =فَالنَّارُ تَحْرِقُنِي مِنَ الشَّوْقِ، وَالْمَاءُ يُطْفِئُنِي مِنَ الدَّمْعِ=. =فَأَنَا بَيْنَ حَرٍّ وَبَرْدٍ، وَبَيْنَ صَبْرٍ وَنَفَادِ صَبْرٍ=. =فَإلَى مَتَى هَذَا الْعَذَابُ؟ وَهَلْ مِنْ وِصَالٍ يُرْجَى؟="

=

> "=اللَّيْلُ عِنْدِي نَهَارٌ، وَالنَّهَارُ عِنْدِي لَيْلٌ=. =فَأَنَا أَسْهَرُ إذَا نَامَ النَّاسُ، وَأَنَامُ إذَا اسْتَيْقَظُوا=. =فَمَا أَنَا مِنْ هَذَا الْعَالَمِ، وَلَا هُمْ مِنِّي=."

=

هل تريد بعد ذلك الانتقال إلى =الأمثال والحكم الشعرية القصيرة (التي تُجرى مجرى المثل)= أم إلى =القصائد المبنية على التضاد من أولها إلى آخرها=؟

تفضّل، إليك =الأمثال والحكم الشعرية القصيرة= التي تجري مجرى المثل، ثم =القصائد الكاملة المبنية على التضاد=، وكلها بالفصحى:

=

== 🪙 أمثال وحكم شعرية قصيرة تجري مجرى المثل

= من الشعر الجاهلي:

=زهير بن أبي سلمى:=
> وَمَن يَجعَلِ المَعروفَ مِن دونِ عِرضِهِ
> يَفِرهُ وَمَن لا يَتَّقِ الشَتمَ يُشتَمِ

> وَمَن يَكُ ذا فَضلٍ فَيَبخَل بِفَضلِهِ
> عَلى قَومِهِ يُستَغنَ عَنهُ وَيُذمَمِ

=

=النابغة الذبياني:=
> وَلَستَ بِمُستَبقٍ أَخاً لا تَلُمُّهُ
> عَلى شَعَثٍ أَيُّ الرِجالِ المُهَذَّبُ

=

=طرفة بن العبد:=
> إِذا كُنتَ في حاجَةٍ مُرسِلاً
> فَأَرسِل حَكيماً وَلا توصِهِ

> وَظُلمُ ذَوي القُربى أَشَدُّ مَضاضَةً
> عَلى المَرءِ مِن وَقعِ الحُسامِ المُهَنَّدِ

=

=عنترة بن شداد:=
> لا يَحمِلُ الحِقدَ مَن تَعلو بِهِ الرُتَبُ
> وَلا يَنالُ العُلا مَن طَبعُهُ الغَضَبُ

=

= من الشعر الأموي والعباسي:

=أبو الأسود الدؤلي:=
> لا تَنهَ عَن خُلُقٍ وَتَأتي مِثلَهُ
> عارٌ عَلَيكَ إِذا فَعَلتَ عَظيمُ

=

=الفرزدق:=
> أَلَيسَ أَبي فينا الَّذي مِن نِزالِهِ
> إِذا الخَيلُ جالَت في النِزالِ مُعَلَّمُ

> هُمُ القَومُ إِن قالوا أَصابوا وَإِن دُعوا
> أَجابوا وَإِن أَعطَوا أَطابوا وَأَجزَلوا

=

=جرير:=
> إِنَّ العُيونَ الَّتي في طَرفِها حَوَرٌ
> قَتَلنَنا ثُمَّ لَم يُحيِينَ قَتلانا

=

=أبو تمام:=
> السَيفُ أَصدَقُ أَنباءً مِنَ الكُتُبِ
> في حَدِّهِ الحَدُّ بَينَ الجِدِّ وَاللَعِبِ

> نَقِّل فُؤادَكَ حَيثُ شِئتَ مِنَ الهَوى
> ما الحُبُّ إِلّا لِلحَبيبِ الأَوَّلِ

=

=البحتري:=
> عَلَى مَرِّ اللَيالي الدَهرُ يُدنيكَ لِلمَدى
> وَيَبعُدُ مِن آمالِ ما كُنتَ تَطلُبُ

=

=أبو الطيب المتنبي:=
> إِذا أَنتَ أَكرَمتَ الكَريمَ مَلَكتَهُ
> وَإِن أَنتَ أَكرَمتَ اللَئيمَ تَمَرَّدا

> لا تَشتَرِ العَبدَ إِلّا وَالعَصا مَعَهُ
> إِنَّ العَبيدَ لَأَنجاسٌ مَناكيدُ

> ما كُلُّ ما يَتَمَنّى المَرءُ يُدرِكُهُ
> تَجري الرِياحُ بِما لا تَشتَهي السُفُنُ

> وَلَم أَرَ كَالأَخلاقِ حَلياً لِفاحِمٍ
> وَلَم أَرَ كَالفِعلِ الجَميلِ جَمالا

=

=أبو العلاء المعري:=
> أَلا في سَبيلِ المَجدِ ما أَنا فاعِلُ
> سَأَجعَلُ بَينَ العُذرِ وَاللَومِ فاصِلا

> وَإِنّي وَإِن كُنتُ الأَخيرَ زَمانُهُ
> لَآتٍ بِما لَم تَستَطِعهُ الأَوائِلُ

=

=الشريف الرضي:=
> أُعَلِّلُ نَفسي بِالأَماني وَأَبتَغي
> شِفاءً لَها وَالشَوقُ نَحوَكَ قاتِلي

=

= من الشعر الأندلسي:

=ابن زيدون:=
> بنتم وَبِنّا فَما اِبتَلَّت جَوانِحُنا
> شَوقاً إِلَيكُم وَلا جَفَّت مَآقينا

> إِنّي ذَكَرتُكِ بِالزَهراءِ مُشتاقاً
> وَالأُفقُ طَلقٌ وَمَرأى الأَرضِ قَد راقا

=

=المعتمد بن عباد:=
> قَيدٌ يُقَيِّدُني عَن كُلِّ مَكرُمَةٍ
> وَالدَهرُ يُوسِعُني ضَرّاً وَإِرهاقا

=

=ابن خفاجة:=
> يا أَهلَ أَندَلُسٍ لِلَّهِ دَرُّكُمُ
> ماءٌ وَظِلٌّ وَأَنهارٌ وَأَشجارُ

=

= من الشعر الحديث:

=أحمد شوقي:=
> إِنَّما الأُمَمُ الأَخلاقُ ما بَقِيَت
> فَإِن هُمُ ذَهَبَت أَخلاقُهُم ذَهَبوا

> قُم لِلمُعَلِّمِ وَفِّهِ التَبجيلا
> كادَ المُعَلِّمُ أَن يَكونَ رَسولا

=

=حافظ إبراهيم:=
> الأُمُّ مَدرَسَةٌ إِذا أَعدَدتَها
> أَعدَدتَ شَعباً طَيِّبَ الأَعراقِ

=

=إيليا أبو ماضي:=
> كَم ذا يُكابِدُ عاشِقٌ وَيُلاقي
> في حُبِّ مِصرَ كَثيرَةِ العُشّاقِ

> أَنا لا أَقولُ إِذا المَواسِمُ أَقبَلَت
> هَذا رَبيعٌ بَل أَقولُ خَريفُ

=

=محمود درويش:=
> سَجِّل أَنا عَرَبِي
> وَرَقَمُ بَطاقَتي خَمسونَ أَلف

=

=نزار قباني:=
> أَتُراني أَحبَبتُ حَتّى بَلَغَ الحُ
> بُّ دَراكاً فَصارَ نَوعاً مِنَ الكُفرِ

> فَارَقْتَنِي فَارْتَحَلَ الصَّبَاحُ
> وَارْتَحَلَ المَرَفُ وَالأَرِيْكَةْ

=

= حكم شعرية قصيرة مفردة (أبيات مفردة سائرة):

> =تَجْرِي الرِّيَاحُ بِمَا لَا تَشْتَهِي السُّفُنُ=
> (المتنبي)

> =لِكُلِّ شَيْءٍ إِذَا مَا تَمَّ نُقْصَانُ=
> (أبو البقاء الرندي)

> =وَمَا نَيْلُ الْمَطَالِبِ بِالتَّمَنِّي=
> =وَلَكِنْ تُؤْخَذُ الدُّنْيَا غِلَابَا=
> (أبو القاسم الشابي)

> =إِذَا الشَّعْبُ يَوْمًا أَرَادَ الْحَيَاةَ=
> =فَلَا بُدَّ أَنْ يَسْتَجِيبَ الْقَدَر=
> (أبو القاسم الشابي)

> =وَمَنْ يَتَهَيَّبْ صُعُودَ الْجِبَالِ=
> =يَعِشْ أَبَدَ الدَّهْرِ بَيْنَ الْحُفَر=
> (أبو القاسم الشابي)

> =وَلِلْحُرِّيَّةِ الْحَمْرَاءِ بَابٌ=
> =بِكُلِّ يَدٍ مُضَرَّجَةٍ يُدَقُّ=
> (أبو القاسم الشابي)

> =كُنْ بَلْسَمًا إِنْ دَامَ جُرْحُ قَبِيلَةٍ=
> =أَوْ لَا فَكُنْ عَوْنًا عَلَى الْبَلْسَمَةِ=
> (إيليا أبو ماضي)

> =لَسْتُ أَدْرِي أَقُبَيْلَ الصُّبْحِ أَمْسِي=
> =أَمْ أَمَامَ اللَّيْلِ هَذَا أَمْ وَرَائِي=
> (عمر أبو ريشة)

=

== 📜 قصائد كاملة مبنية على التضاد

= القصيدة الأولى: قصيدة أبي البقاء الرندي في رثاء الأندلس

> لِكُلِّ شَيْءٍ إِذَا مَا تَمَّ نُقْصَانُ
> فَلَا يُغَرَّ بِطِيبِ العَيْشِ إِنْسَانُ
> هِيَ الأُمُورُ كَمَا شَاهَدْتُهَا دُوَلٌ
> مَنْ سَرَّهُ زَمَنٌ سَاءَتْهُ أَزْمَانُ
> وَهَذِهِ الدَّارُ لَا تُبْقِي عَلَى أَحَدٍ
> وَلَا يَدُومُ عَلَى حَالٍ لَهَا شَانُ
> =يُمَزِّقُ الدَّهْرُ حَتْمًا كُلَّ سَابِغَةٍ=
> =إِذَا نَبَتْ مَشْرَفِيَّاتٌ وَخُرْصَانُ=
> وَيَنْتَضِي كُلُّ سَيْفٍ لِلْفَنَاءِ وَلَوْ
> كَانَ اِبْنَ ذِي يَزَنٍ وَالغِمْدَ غُمْدَانُ
> أَيْنَ المُلُوكُ ذَوُو التِّيجَانِ مِنْ يَمَنٍ
> وَأَيْنَ مِنْهُمْ أَكَالِيلٌ وَتِيجَانُ
> وَأَيْنَ مَا شَادَهُ شَدَّادُ فِي إِرَمٍ
> وَأَيْنَ مَا سَاسَهُ فِي الْفُرْسِ سَاسَانُ
> وَأَيْنَ مَا حَازَهُ قَارُونُ مِنْ ذَهَبٍ
> وَأَيْنَ عَادٌ وَشَدَّادٌ وَقَحْطَانُ
> أَتَى عَلَى الكُلِّ أَمْرٌ لَا مَرَدَّ لَهُ
> حَتَّى قَضَوْا فَكَأَنَّ القَوْمَ مَا كَانُوا
> =وَصَارَ مَا كَانَ مِنْ مُلْكٍ وَمِنْ مَلِكٍ=

= القصيدة الثانية: قصيدة المتنبي في وصف الحمى (طباق متواصل)

= القصيدة الثالثة: قصيدة أبي فراس الحمداني في الأسر (طباق العز والذل)

> =وَلَكِنَّنِي فِي الْحُبِّ أَهْوَنُ مَا بِيَا=

= القصيدة الرابعة: قصيدة "الموت في الحياة" لأبي العلاء المعري

> =الحَيُّ مَيْتٌ وَإِنْ طَالَتْ سَلَامَتُهُ=
> =وَالْمَيْتُ حَيٌّ إِذَا مَا ذِكْرُهُ بَقِيَا=
> =فَاخْتَرْ لِنَفْسِكَ مَا تَبْقَى مَحَاسِنُهُ=
> =إِنَّ الْفَتَى بَعْدَ مَوْتِ الْجِسْمِ مَرْئِيَّا=
> =لَا تَحْسَبِ الْمَوْتَ مَوْتاً إِنَّمَا هُوَ فِي=
> =حَيَاةِ أُخْرَى لِمَنْ قَدْ كَانَ مَرْضِيَّا=
> =فَالنَّاسُ بَيْنَ حَيَاةٍ ثُمَّ مَيْتَتِهِ=
> =وَبَيْنَ مَوْتٍ إِذَا مَا اسْتَوْطَنَ الْحَيَّا=
> =لَا تَطْلُبَنَّ بِأَمْرٍ لَسْتَ مُدْرِكَهُ=

> =ثُمَّ انْقَضَى فَكَأَنْ لَمْ يُلْفَ مَوْجُودَا=

= القصيدة الخامسة: قصيدة "إرادة الحياة" لأبي القاسم الشابي (طباق متواصل)

> =إِذَا الشَّعْبُ يَوْمًا أَرَادَ الْحَيَاةَ=
> =فَلَا بُدَّ أَنْ يَسْتَجِيبَ الْقَدَرْ=
> =وَلَا بُدَّ لِلَّيْلِ أَنْ يَنْجَلِي=

> مِنْ صَفْعَةِ الْعَدَمِ الْمُنْتَصِرْ
> كَذَلِكَ قَالَتْ لِيَ الْكائناتُ
> وَحَدَّثَنِي رُوحُهَا الْمُسْتَتِرْ
> =وَمَنْ يَتَهَيَّبْ صُعُودَ الْجِبَالِ=

> =فَعَجَّتْ بِقَلْبِي دِمَاءُ الشَّبَابِ=
> =وَخَفَّتْ بِصَدْرِي رِيَاحٌ أُخَرْ=
> وَأَطْرَقْتُ أُصْغِي لِقَصْفِ الرُّعُودِ
> وَعَزْفِ الرِّيَاحِ وَوَقْعِ الْمَطَرْ
> =وَقَالَتْ لِيَ الأَرْضُ لَمَّا سَأَلْتُ:=
> =أَيَا أُمُّ هَلْ تَكْرَهِينَ الْبَشَرْ؟=
> =أُبَارِكُ فِي النَّاسِ أَهْلَ الطُّمُوحِ=
> =وَمَنْ يَسْتَلِذُّ رُكُوبَ الْخَطَرْ=
> =وَأَلْعَنُ مَنْ لَا يُمَاشِي الزَّمَانَ=
> =وَيَقْنَعُ بِالْعَيْشِ عَيْشِ الْحَجَرْ=

> وَيَحْتَقِرُ الْمَيْتَ الْمُحْتَقَرْ
> =فَلَا الأُفُقُ الْيَوْمَ يَحْوِي النُّجُومَ=
> =وَلَا الْقَمَرُ اللَّيْلَ بَدْرٌ أَغَرْ=
> =وَلَا الْبَحْرُ بَحْرٌ وَلَا الْبَرُّ بَرٌّ=
> =إِذَا لَمْ يَكُنْ فِي ضَمِيرِ الْبَشَرْ=
> =فَمَنْ لَمْ يَشُقْهُ جَلَالُ الْحَيَاةِ=
> =تَبَخَّرَ فِي جَوِّهَا وَانْدَثَرْ=

> =فَسُبْحَانَ مَنْ سَوَّى النَّهَارَ وَزَانَهُ=
> =وَسُبْحَانَ مَنْ سَوَّى الظَّلَامَ وَأَغْدَقَا=
> =وَجَعَلَ فِي الْأَضْدَادِ سِرَّ تَمَامِهِ=
> =فَمَا طَابَ لَوْلَا مُرُّهُ لِيُذَاقَا=

(طباق في: ليل / صبح – صمت / نور – تضيء / ظلام – ضحى / دجنة – ليل / نهار – أضداد / تمام – طاب / مر)

=

= القصيدة العاشرة: قصيدة "الغنى والفقر" لعلي بن أبي طالب (منسوبة)

> =النَّفْسُ تَبْكِي عَلَى الدُّنْيَا وَقَدْ عَلِمَتْ=
> =أَنَّ السَّلَامَةَ فِيهَا تَرْكُ مَا فِيهَا=
> لَا دَارَ لِلْمَرْءِ بَعْدَ الْمَوْتِ يَسْكُنُهَا
> إِلَّا الَّتِي كَانَ قَبْلَ الْمَوْتِ بَانِيهَا
> =فَإِنْ بَنَاهَا بِخَيْرٍ طَابَ مَسْكَنُهَا=
> =وَإِنْ بَنَاهَا بِشَرٍّ خَابَ بَانِيهَا=
> أَيْنَ الْمُلُوكُ الَّتِي كَانَتْ مُسَلْطَنَةً
> حَتَّى سَقَاهَا بِكَأْسِ الْمَوْتِ سَاقِيهَا
> أَمْوَالُنَا لِذَوِي الْمِيرَاثِ نَجْمَعُهَا
> وَدُورُنَا لِخَرَابِ الدَّهْرِ نَبْنِيهَا
> =كَمْ مِنْ غَنِيٍّ أَتَى الدُّنْيَا فَعَاشَ بِهَا=
> =ثُمَّ انْقَضَى وَكَأَنْ لَمْ يَلْقَ مَا فِيهَا=
> =وَكَمْ فَقِيرٍ تَعَالَى النَّفْسُ عَاشَ بِهَا=
> =حَتَّى مَضَى وَالرِّضَا فِي وَجْهِهِ زَاهِي=
> =فَالْفَقْرُ فِي النَّفْسِ لَا فِي الْمَالِ نَعْرِفُهُ=
> =وَالْغِنَى فِي النَّفْسِ لَا فِي الْمَالِ يَا سَاهِي=

(طباق في: خير / شر – غني / فقير – فقر / غنى – نفس / مال)

=

هل تريد بعدها الانتقال إلى =أشعار المديح والهجاء المبنية على التضاد= (كشعر النقائض بين جرير والفرزدق)، أم إلى =الأشعار الصوفية= التي يكثر فيها الطباق بين الظاهر والباطن؟

سأجمع لك الآن =النقائض= (شعر المديح والهجاء بالتضاد) بين جرير والفرزدق وغيرهما، ثم =الأشعار الصوفية= المبنية على الطباق بين الظاهر والباطن، وكلها بالفصحى:

=

== ⚔️ النقائض: شعر المديح والهجاء المبنية على التضاد

النقائض هي قصائد يقولها شاعر فيمدح نفسه وقبيلته ويهجو خصمه، فيرد الآخر بقصيدة على الوزن والقافية نفسيهما مادحاً نفسه هاجياً الأول. وأشهرها =نقائض جرير والفرزدق=.

=

= نماذج من نقائض جرير والفرزدق

=قال جرير يهجو الفرزدق ويمدح قومه:=

> أَلَستُم خَيرَ مَن رَكِبَ المَطايا
> وَأَندى العالَمينَ بُطونَ راحِ
> =قَوَمٌ أَبوهُم أَبو العاصي وَقَومي=
> =هُمُ الأَعزونَ إِن ذُكِرَ القِباحُ=
> =فَإِن تَكُ قَد مُنيتَ بِنا فَإِنّا=
> =مَنَخناكَ الهَوانَ بِلا جِراحِ=
> =وَقَومي إِن سَأَلتَ هُمُ نِزاري=
> =وَقَومُكَ إِن سَأَلتَ هُمُ الصُداحُ=

(طباق في: خير / أقبح – أعز / هوان – قومي / قومك – نزار / صداح (قبائل متضادة في السياق))

=

=فأجابه الفرزدق:=

> =أَلَستُم نَهشَلاً وَهُمُ لِئامٌ=
> =وَنَحنُ الأَكرَمونَ أَبي وَراحي=
> =فَإِن تَكُ قَد فَخَرتَ بِأُمِّ سَعدٍ=
> =فَإِنّا قَد غَلَبناكَ بِالرِّماحِ=
> =فَلا تَفخَر بِخِندِفَ إِنَّ قَومي=
> =بَنو تَميمٍ هُمُ الأَعلَونَ راحي=
> =وَقَومي إِن سَأَلتَ هُمُ تَميمٌ=
> =وَقَومُكَ إِن سَأَلتَ هُمُ القِباحُ=

(طباق في: لئام / أكرمون – خندف / تميم – أعلون / أقبح – قومي / قومك)

=

=وقال جرير في هجاء الفرزدق:=

> =فَإِنَّ الفَخرَ بِالماضي مَعادٌ=
> =وَلَيسَ الفَخرُ إِلّا ما بَنَيتُم=
> =وَقَومي الفاتِحونَ لِكُلِّ بابٍ=
> =وَقَومُكَ ما فَتَحتُم مِن سِدادِ=
> =لَنا الدُنيا إِذا سُدنا وَقامَت=
> =وَنَحنُ النازِلونَ بِكُلِّ وادي=
> =وَقَومي المُنعِمونَ إِذا اِشتَكَينا=
> =وَقَومُكَ الظالِمونَ مَعَ الفَسادِ=

(طباق في: ماض / ما بنيتم – فاتحون / ما فتحتم – منعمون / ظالمون)

=

=فأجابه الفرزدق:=

> =فَإِنَّ المَجدَ لَيسَ لَهُ مَعادٌ=
> =إِذا ما ضاعَ في نَسبِ العَبيدِ=
> =وَقَومي الفارِعونَ ذُرى المَعالي=
> =وَقَومُكَ النازِلونَ عَلى الوَهيدِ=
> =لَنا العَلياءُ في الدُنيا جَميعاً=
> =وَنَحنُ الطاعِنونَ بِكُلِّ زَندِ=
> =وَقَومي المُكرِمونَ لِكُلِّ ضَيفٍ=
> =وَقَومُكَ الباخِلونَ عَلى الوَفودِ=

(طباق في: ماض / ضاع – فارعون / نازلون – مكرمون / باخلون)

=

=وقال جرير يهجو الفرزدق بأشنع ما قيل فيه:=

> =أَلَم تَرَ أَنَّني وَالموتَ أَخوي=
> =وَلَيسَ لِغَيرِنا في الأَمرِ نَفسُ=
> =فَإِن أَهلِك فَرُبَّ فَتًى كَرِيمٍ=
> =يُصابُ وَمِثلُ رُزئِهِ تَأَسُّ=
> =وَقَد أَبقَت لَنا الأَيّامُ مِنهُ=
> =عَلى اللَأواءِ باقِيَةً تَدُسُّ=

=

=فأجابه الفرزدق بأقسى ما قيل في الهجاء:=

> =لَقَد وَلَدَت أُمُّ الفَرَزدَقِ فاجِراً=
> =فَجاءَت بِوَزنِ الفيلِ في خَلقِ نَملَةِ=
> =أَتَغضَبُ أَن يُهدى الخَنافِرُ حَولَنا=
> =وَمِنكَ بِما تُخفيهِ حَولَ مَباءَةِ=

(طباق في: فيل / نملة – ضخامة في الذم / حقارة في الخلقة)

=

= نماذج من نقائض الأخطل وجرير

=قال الأخطل يهجو جريراً ويمدح قومه:=

> =أَلا مَن مُبلِغٌ جَريراً رِسالاً=
> =فَيَعلَمَ أَنَّني صَقرٌ طَليقُ=
> =وَأَنَّكَ يا جَريرُ أَخَسُّ كَلبٍ=
> =يُشَبَّهُ في المَهانَةِ وَالضُّيوفِ=
> =فَقَومي تَغلِبُ الغَرّاءُ سادوا=
> =وَقَومُكَ بَينَ خاسِئَةٍ وَجُوفِ=
> =لَنا المَجدُ المُؤَثَّلُ وَالعَطايا=
> =وَعِندَكُمُ المَذَلَّةُ وَالحُتوفُ=

(طباق في: صقر / كلب – سادوا / خاسئة – مجد / مذلة – عطايا / حتوف)

=

=فأجابه جرير:=

> =أَلا مَن مُبلِغٌ تَغلِبَ رِسالاً=
> =بِأَنَّ اللُومَ مَورِدُها العَميقُ=
> =فَإِن تَكُ تَغلِبُ الغَرّاءُ قالَت=
> =فَقَد كَذَبَت وَأَنتَ بِها صَدوقُ=
> =فَقَومي بَيتُ أَولِ مَن تَسَمّى=
> =وَقَومُكَ كُلُّهُم عَبدٌ رَقيقُ=
> =لَنا البَطحاءُ تَملَؤُها قِباباً=
> =وَعِندَكُمُ الخَمائِلُ وَالسُّعوقُ=

(طباق في: كذبت / صدوق – بيت أول / عبد رقيق – بطحاء / خمائل)

=

= نماذج من نقائض أخرى (بين بشار بن برد وحماد عجرد)

=قال بشار يهجو حماداً:=

> =أَيا حَمّادُ أَنتَ أَخَسُّ خَلقٍ=
> =وَأَنتَ أَذَلُّ مَن وَطِئَ التُرابا=
> =فَإِنَّكَ لِلنِساءِ غَداةَ تَبدو=
> =كَقِردٍ زانَ أَكتافاً وَغابا=
> =وَلَو أَنَّ الجَمالَ يُباعُ يَوماً=
> =لَما اِشتَرى أَحَدٌ لَكَ بِالتُرابِ=

=

=فأجابه حماد:=

> =أَيا بَشّارُ أَنتَ أَشَدُّ كُفراً=
> =وَأَنتَ أَضَلُّ مَن عَبَدَ الصَليبا=
> =فَإِنَّكَ أَعوَرُ العَينَينِ فَدمٌ=
> =تَعيبُ النّاسَ تَضحَكُ وَالغِيابا=
> =وَلَو أَنَّ العُيوبَ تُباعُ يَوماً=
> =لَكُنتَ أَعَزَّ مُنقَطِعٍ طِلابا=

(طباق في: أخس / أعز – جمال / عيوب – اشتري / تباع)

=

== 🌌 الأشعار الصوفية المبنية على التضاد (الظاهر والباطن)

الصوفية يكثرون من الطباق بين المعاني الظاهرة والباطنة؛ فيجمعون بين النور والظلمة، والحضور والغيبة، والفناء والبقاء، والموت والحياة، والوصل والفراق.

=

= من شعر ابن الفارض (سلطان العاشقين):

> =شَرِبنا عَلى ذِكرِ الحَبيبِ مُدامَةً=
> =سَكِرنا بِها مِن قَبلِ أَن يُخلَقَ الكَرمُ=
> =لَها البَدرُ كَأسٌ وَالشُموسُ دَوارِقٌ=
> =وَهِلالٌ لَها نَثَرَ النُجومَ وَقَد ضَمُّ=
> =فَلا خَمرَةٌ تَحكي لَدَيها مُدامَتي=
> =وَلا سَكرَةٌ مِثلُ المُدامَةِ تَنهَزِمُ=

(طباق في: سكرنا / قبل أن يخلق الكرم – بدر / هلال – شموس / نجوم)

=

> =أَنا في الحَقيقَةِ لا أَنا=
> =بَل نَحنُ نَحنُ وَلا نَحنُ=
> =قَد زالَ عَنّي رَسمُ نَفسي=
> =وَاِضمَحَلَّتِ النَفسُ وَالظَنُّ=
> =فَصَحَوتُ مِن سُكري بِسُكرٍ=
> =وَاِنقَضَتِ الأَوهَامُ وَالغَبنُ=
> =فَالبَحرُ مُنتَصِبٌ بِقَلبي=
> =وَالنارُ مُضطَرِمٌ لَها سَكنُ=

(طباق في: أنا / لا أنا – نحن / لا نحن – صحوت / سكر – بحر / نار – مضطرم / سكن)

=

> =فَيا وَيحَ قَلبٍ ضاقَ بي وَاِتَّسَعتُمُ=
> =فَكَيفَ يَضيقُ القَلبُ عَمّا أَقولُهُ=
> =وَيا عَجَبا مِن قُربِكُم وَاِبتِعادِكُم=
> =وَمِن حُبِّكُم رَغمَ الجَفاءِ وَمِنهُم=
> =أُحِبُّكُمُ حُبَّينِ: حُبّاً لِقُربِكُم=
> =وَحُبّاً لِأَن أَهلَ الوَفاءِ قَليلُ=

(طباق في: ضاق / اتسعتم – قربكم / ابتعادكم – حبين / قليل)

=

= من شعر رابعة العدوية (رحمها الله):

> =أُحِبُّكَ حُبَّينِ: حُبَّ الهَوى=
> =وَحُبّاً لِأَنَّكَ أَهلٌ لِذاكا=
> =فَأَمّا الَّذي هُوَ حُبُّ الهَوى=
> =فَشُغلي بِذِكرِكَ عَمَّن سِواكا=
> =وَأَمّا الَّذي أَنتَ أَهلٌ لَهُ=
> =فَكَشفُكَ لِلحُجبِ حَتّى أَراكا=
> =فَلا الحَمدُ في ذا وَلا ذاكَ لي=
> =وَلَكِن لَكَ الحَمدُ في ذا وَذاكا=

(طباق في: حب الهوى / حب لأنك أهل – ذا / ذاك – لي / لك)

=

> =يا سُروري وَمُنيَةَ القَلبِ نَفسي=
> =أَنتَ لي مُؤنِسي إِذا أَوَحَشَتني=
> =إِن تَكُن راضِياً فَإِنّي بِسَعدٍ=
> =أَو تَكُن ساخِطاً فَفي القَلبِ يَأسُ=

(طباق في: راضياً / ساخطاً – سعد / يأس)

=

= من شعر أبي الحسن الششتري:

> =أَنا الفَقيرُ إِلَيكَ في حالي=
> =وَأَنتَ غَنِيٌّ عَن كُلِّ أَمثالي=
> =يا مَن يَراني وَلا أَراهُ سِوى=
> =في كُلِّ شَيءٍ لَهُ تَجَلّى لي=
> =إِن غِبتَ عَن ناظِري فَأَنتَ مَعي=
> =في السِرِّ مِنّي وَفي جَميعِ الخَطَرْ=
> =فَلا حَياةٌ بِدونِ قُربِكَ لي=
> =وَلا مَماتٌ إِذا دَنَوتَ ضَرَرْ=

(طباق في: فقير / غني – يراني / لا أراه – غبت / معي – حياة / ممات)

=

= من شعر النفري (صاحب المواقف والمخاطبات):

> =أَوقَفَني في العِزِّ وَقالَ لي:=
> =العِزُّ أَن تَذِلَّ لِي، وَالذُلُّ أَن تَعتَزَّ بِسِوايَ.=
> =وَوَقَفَني في الغِنى وَقالَ لي:=
> =الغِنى أَن تَفتَقِرَ إِلَيَّ، وَالفَقرُ أَن تَستَغنِيَ عَنّي.=
> =وَوَقَفَني في الحَياةِ وَقالَ لي:=
> =الحَياةُ أَن تَموتَ فيَّ، وَالمَوتُ أَن تَحيا بِغَيري.=

(طباق في: عز / ذل – غنى / فقر – حياة / موت – تموت في / تحيا بغيري)

=

= من شعر الحلاج أيضاً (في الوحدة والتعدد):

> =أَنتَ بَينَ الشَغافِ وَالقَلبِ تَجري=
> =مَجرَى الدَمعِ مِن جُفونِ العُيونِ=
> =وَأَنتَ الأَولِياءُ في الحَشرِ أَنتَ الـ=
> =ـحَشرُ يَوماً وَأَنتَ في كُلِّ حينِ=
> =تَتَجَلّى لِمَن تُحِبُّ وَتَخفى=
> =فَتَرى الناسَ في العَمى وَاليَقينِ=
> =فَأَنا الغائِبُ الحَضورُ وَفيكُم=
> =أَنا أَبكي عَلى النَوى وَالحَنينِ=

(طباق في: تجري / الدمع – تتجلى / تخفى – عمى / يقين – غائب / حضور – أبكي / حنين)

=

= من شعر محيي الدين بن عربي (في الفناء والبقاء):

> =فَنيتُ عَن رُؤيَتي فَأَبقَيتَني=
> =وَمُتُّ عَن حِسّي فَأَحيَيتَني=
> =وَغِبتُ عَن عِلمي فَأَنتَ الَّذي=
> =تُديرُني حَيثُما شِئتَني=
> =فَأَنتَ سَمعي إِذا ما اِستَمَعتُ=
> =وَأَنتَ بَصَري حينَ أَبصَرتُني=
> =فَلا أَنا أَنا وَلا أَنتَ أَنتَ=
> =كَأَنَّني أَنتَ وَأَنتَ أَنا=

(طباق في: فنيت / أبقيتني – مت / أحييتني – غبت / تديرني – أنا أنا / أنت أنت – أنت أنا / أنا أنت)

=

= من شعر عبد القادر الجيلاني (منسوب):

> =يا غافِلاً ما آنَ أَن تَستَيقِظا=
> =فَاللَيلُ وَلّى وَالصَباحُ تَلَألَأا=
> =يا طالِبَ الدُنيا وَزُخرُفَها اسْمَعَن=
> =مِنّي الحَقيقَةَ لا تَخَف أَن تَظلَما=
> =الدُنيا حَلاوَتُها يَعقُبُها أَذىً=
> =وَعِزُّها ذُلٌّ لِمَن قَد أَجرَما=
> =وَفَقيرُها مَن باتَ يَجمَعُ مالَها=
> =وَغَنِيُّها مَن باتَ فيها مُعدَما=

(طباق في: غافل / يستيقظ – ليل / صباح – حلاوة / أذى – عز / ذل – فقير / غني – مال / معدما)

=

= من شعر البوصيري الصوفي (البردة):

> =مُحَمَّدٌ سَيِّدُ الكَونَينِ وَالثَقَلَيـ=
> =ـنِ وَالفَريقَينِ مِن عُربٍ وَمِن عَجَمِ=
> =هُوَ الحَبيبُ الَّذي تُرجى شَفاعَتُهُ=
> =لِكُلِّ هَولٍ مِنَ الأَهوالِ مُقتَحَمِ=
> =ثُمَّ الرِّضا عَن أَبي بَكرٍ وَعَن عُمَرٍ=
> =وَعَن عُثمانَ ذي النورَينِ وَالكَرَمِ=
> =وَعَن عَلِيٍّ أَميرِ المُؤمِنينَ وَمَن=
> =والاهُمُ صادِقاً في السِرِّ وَالعَلَنِ=

(طباق في: سر / علن – عرب / عجم – كونين / فريقين)

=

هل تريد بعدها الانتقال إلى =الأشعار التعليمية المطولة= (كألفية السيوطي أو منظومات في الأضداد)، أم العودة إلى =توسيع أي قسم مما سبق=؟

تفضّل، إليك =منظومات تعليمية مطوّلة في الأضداد=، تشمل ما ورد في كتب التراث اللغوي والأدبي، وكلها بالفصحى:

=

== 📘 منظومات في الأضداد من كتب التراث

= المنظومة الأولى: منظومة ابن السكيت في الأضداد (مشهورة ومتداولة)

> =وَمِن عَجَبِ الأَضْدَادِ أَنَّكَ تَلتَقِي=
> بِمَعنًى وَضِدِّ المَعنِيَينِ جَمِيعا
> =فَمِن ذَاكَ (جَوْنٌ) لِلظَّلامِ وَلِلضُّحى=
> وَ(صَريمَةٌ) لِلَّيلِ وَالصُّبحِ تَطْلُعُ
> =و(ظَنٌّ) لِمَعنَى الشَّكِّ وَاليَقينِ مَعاً=
> وَ(سَدَفٌ) لِلظَّلْماءِ وَالنّورِ يَلْمَعُ
> =و(عَجفَاءُ) لِلحَسْناءِ وَلشَّوْهَاءِ مَعاً=
> وَ(رَهوَةٌ) لِلوَهدِ وَالرَّبْوَةِ اِجمَعِ
> =و(رَجَبٌ) لِلتَّعظِيمِ وَالشَّتْمِ سَوِّيَت=
> حُرُوفُهُ فَافهَم ما أَقُولُ لِتَسمَعِ

=

=وتكملتها:=

> =و(جَلَلٌ) لِليَسيرِ وَلِلعَظيمِ مَعاً=
> وَ(ثَوى) لِلِاستِقرارِ وَالسَّيرِ أَسرَعِ
> =و(عَسعَسَ) اللَّيلُ إِذ أَقبَلتَ تَرقُبُهُ=
> وَإِذ أَدبَرتَ هَذا القَولُ فَاِستَمِعِ
> =و(قَضى) لِلقَضاءِ وَلِلمَوتِ الَّذي نَزَلَت=
> بِهِ المَنِيَّةُ فَاِعلَم ذاكَ وَاِنتَفِعِ
> =و(حَشا) لِلجَوفِ وَلِلنّاحِيَةِ اِتَّفَقَت=
> عَلَيهِ أَقوالُ أَهلِ الفَضلِ في اللُّمَعِ

=

= المنظومة الثانية: منظومة أبي الطيب اللغوي في الأضداد

> =تَأَمَّلْ فِي كَلَامِ الْعَرَبِ تَجِدْ=
> عَجَائِبَ فِي الْأَضْدَادِ تُرْوَى وَتُعْتَمَدْ
> =فَمِنْهَا (الْجَوْنُ) يَأْتِي لِلظَّلَامِ وَلِلضُّحَى=
> وَ(الصَّرِيمُ) لِلَيْلٍ وَلِلصُّبْحِ إِنْ وُجِدْ
> =وَمِنْهَا (السَّدِيفُ) الشَّحْمُ وَالسَّدِيفُ فِي=
> مَعَانِيهِمُ أَيْضاً لِمَا فَاتَ مِنْ عَهْدِ
> =وَمِنْهَا (الْوَلِيُّ) العَدُوُّ وَحَارِسِي=
> وَصَدِيقِيَ الْمَحْبُوبُ فَافْطُنْ لِمَا تَجِدْ
> =وَمِنْهَا (الْعَدُوُّ) الصَّدِيقُ وَضِدُّهُ=
> كَذَاكَ رَوَوْهُ فِي الْأَحَادِيثِ وَالسَّنَدْ

=

> =وَمِنْهَا (السَّجِيَّةُ) لِلطَّبِيعَةِ وَالنَّقِيـ=
> ـضِ كَذَاكَ قَدْ قَالُوهُ فِي بَعْضِ مَا وَرَدْ
> =وَمِنْهَا (الْغَرِيمُ) لِلْمُطَالِبِ وَلَّذِي=
> عَلَيْهِ الدُّيُونُ فَاعْجَبَنْ لِهَذَا الْمُضْطَّرَدْ
> =وَمِنْهَا (الْمَوَالِي) لِلْعَبِيدِ وَضِدِّهِمْ=
> مِنَ السَّادَةِ الْأَعْلَى مَقَاماً وَمُحْتَدِ
> =وَمِنْهَا (الْبَسِيلُ) لِلْحَلَالِ وَلِلْحَرَا=
> مِ جَاءَ فِي بَعْضِ الرِّوَايَاتِ فَاعْتَمِدْ
> =وَمِنْهَا (الْقَرُونُ) لِلْعَزِيزِ وَلِلذَّلِيـ=
> ـلِ فَافْهَمْ هُدِيتَ الرُّشْدَ فِي كُلِّ مُضْطَّرَدْ

=

> =وَمِنْهَا (الظَّنِينُ) الْمُتَّهَمُ وَلِلْأَمِيـ=
> ـنِ كَذَاكَ رَوَوْهُ فِي الْأَحَادِيثِ وَالسَّنَدْ
> =وَمِنْهَا (الْخَشَاشُ) لِلْحَيَاةِ وَلِلْحَشَا=
> رَاتِ الَّتِي تَأْوِي إِلَى الْأَرْضِ وَالْوَهَدْ
> =وَمِنْهَا (الْأَسَى) لِلْحُزْنِ وَلْأَسَى شِفَا=
> ءٌ لِلطَّبِيبِ الْمُجْتَبَى أَحْسَنِ مُعْتَقَدْ
> =وَمِنْهَا (الْمُزَنَّدُ) لِلْبَخِيلِ وَلِلْكَرِيـ=
> ـمِ كَذَاكَ قَدْ قَالُوهُ فِي بَعْضِ مَا وَرَدْ
> =وَمِنْهَا (الْعُرَامُ) لِلْكَثِيرِ وَلِلْقَلِيـ=
> ـلِ فَأَيُّ شَيْءٍ فِي الْأَضَادِيدِ لَمْ يُعَدْ

=

= المنظومة الثالثة: منظومة في الأضداد للأصمعي (منسوبة)

> =أَلَا قُلْ لِمَنْ يَهْوَى الْعُلُومَ وَيَبْتَغِي=
> غَرَائِبَهَا فِي كُلِّ بَدْوٍ وَمَحْضَرِ
> =تَعَلَّمْ مِنَ الْأَضْدَادِ مَا قَدْ جَمَعْتُهُ=
> لِتَعْلَمَ أَنَّ الْقَوْلَ ذُو نَظَرَيْنِ
> =فَمِنْهُ: نَشَرْتُ الثَّوْبَ طَيًّا وَطَيَّهُ=
> نَشْراً، وَمِنْهُ: الْبَيْعُ لِلشِّرَى اقْتَنِ
> =وَمِنْهُ: سَتَرْتُ الشَّيْءَ كَشْفاً وَكَشْفُهُ=
> سَتْراً، وَمِنْهُ: الْجِدُّ فِي اللَّعِبِ افْطُنِ
> =وَمِنْهُ: شَكَوْتُ الْهَمَّ سُرّاً وَسِرُّهُ=
> شَكْوَى، وَمِنْهُ: الْحَزْمُ فِي اللَّعِبِ افْطُنِ
> =وَمِنْهُ: رَجَوْتُ الْأَمْرَ خَوْفاً وَخَوْفُهُ=
> رَجْوَى، فَكُنْ بِالْعِلْمِ أَوْفَى مُحَصَّنِ

=

> =وَمِنْهُ: (عَسْعَسَ) اللَّيْلُ إِذَا جَاءَ مُقْبِلاً=
> وَإِذَا أَدْبَرَ اَيْضاً فَهْوَ فِي النَّوْعَيْنِ
> =وَمِنْهُ: (السَّدِيفُ) الشَّحْمُ وَالسَّدِيفُ مَا=
> تَقَطَّعَ مِنْ قِدَمٍ عَلَى الْحَالَيْنِ
> =وَمِنْهُ: (الْعَرِينُ) لِلْغَابَةِ وَالْعَرِينُ لِلْأَرْ=
> ضِ الْخَلَاءِ كَذَاكَ فِي الْقَوْلَيْنِ
> =وَمِنْهُ: (الْمَتُونُ) لِلصُّلْبِ وَالْمَتُونُ لِلْبَطْـ=
> ـنِ الرَّقِيقِ وَهَذَا ضِمْنَ نَوْعَيْنِ
> =وَمِنْهُ: (الْوَعِيدُ) لِلْوَعْدِ الْجَمِيلِ وَلِلْـ=
> ـتَهْدِيدِ فَافْهَمْ أُصُولَ الْعِلْمِ فِي الْفَنَّيْنِ

=

= المنظومة الرابعة: منظومة ابن الدهان النحوي في الأضداد

> =بَدَأْتُ بِحَمْدِ اللَّهِ نَظْمَ قَصِيدَةٍ=
> تُبَيِّنُ مَا فِي الْأَضْدَادِ مِنْ كُلِّ مُشْكِلِ
> =فَمِنْ ذَلِكَ (الْجَوْنُ) الْبَعِيدُ تَطَاوُلاً=
> وَ(الْجَوْنُ) لِلْأَسْوَدِ الْبَهِيمِ الْمُخَيَّلِ
> =وَمِنْ ذَلِكَ (الظَّنُّ) الَّذِي هُوَ شَكُّنَا=
> وَ(الظَّنُّ) لِلْيَقِينِ فِي غَيْرِ مَنْزِلِ
> =وَمِنْ ذَلِكَ (الْعَجْفَاءُ) لِلْحَسْنَاءِ وَهْيَ=
> أَيْضاً لِذَاتِ الشُّوْهِ فِي كُلِّ مَحْفِلِ
> =وَمِنْ ذَلِكَ (الْأَسَى) لِلْحُزْنِ وَالْأَسَى=
> لِمَعْنَى الطِّبَابَةِ عِنْدَ الْحَكِيمِ الْمُؤَمِّلِ

=

> =وَمِنْ ذَلِكَ (الرَّهْوُ) لِلْوَادِي الْمُنْخَفِضِ وَهْوَ=
> أَيْضاً لِمُرْتَفَعِ الْأَرْضِ فِي الْقَوْلِ الْأَجْمَلِ
> =وَمِنْ ذَلِكَ (النَّاهِقُ) لِلْعَظْمِ وَالنَّاهِقُ=
> لِلصَّوْتِ مِنْ حَمِيرِ الْوَعْرِ فِي الْبِيدِ الْمُمَلِّ
> =وَمِنْ ذَلِكَ (الْمَوْلَى) لِمَنْ كَانَ مُنْعِماً=
> وَ(الْمَوْلَى) لِمَنْ بِالْعِتْقِ عَنْهُ تَوَسَّلِ
> =وَمِنْ ذَلِكَ (السَّدَفُ) الظَّلَامُ وَضِدُّهُ=
> مِنَ النُّورِ إِذْ يَبْدُو لِعَيْنِ الْمُؤَمِّلِ
> =وَمِنْ ذَلِكَ (الْقَرْءُ) الطُّهْرُ وَالْقَرْءُ لِلْحَيْـ=
> ـضِ كِلَاهُمَا فِي مُحْكَمِ الذِّكْرِ مُنْزَلِ

=

= المنظومة الخامسة: منظومة تعليمية جامعة في أبواب الفقه (تتضمن الأضداد)

> =وَفِي الْأَضْدَادِ أَبْوَابٌ تُسَرُّ بِهَا النُّفُوسُ=
> إِذَا مَا اسْتَجْمَعَتْ فِي الذِّهْنِ تَأْتِي بِالْعَرُوسِ
> =فَفِي الصَّلَاةِ: (قِيَامٌ ضِدُّهُ قُعُودُ)=
> وَ(رُكُوعٌ ضِدُّهُ رَفْعٌ) كَمَا قَدْ قَالَ هُودُ
> =وَفِي الصِّيَامِ: (صِيَامٌ ضِدُّهُ إِفْطَارُ)=
> وَ(لَيْلٌ ضِدُّهُ نَهَارٌ) لِلصَّائِمِ الْمُخْتَارُ
> =وَفِي الزَّكَاةِ: (غِنًى ضِدُّهُ فَقْرٌ)=
> وَ(إِعْطَاءٌ ضِدُّهُ مَنْعٌ) لِذِي الْكِبْرِ
> =وَفِي الْحَجِّ: (إِحْرَامٌ ضِدُّهُ إِحْلَالُ)=
> وَ(حِلٌّ ضِدُّهُ حَرَمٌ) عَلَى مَا قَالُوا

=

> =وَفِي النِّكَاحِ: (نِكَاحٌ ضِدُّهُ طَلَاقُ)=
> وَ(وَفَاقٌ ضِدُّهُ شِقَاقٌ) وَهَذَا مَا يُطَاقُ
> =وَفِي الْبُيُوعِ: (حَلَالٌ ضِدُّهُ حَرَامُ)=
> وَ(رِبْحٌ ضِدُّهُ خُسْرَانٌ) لِمَنْ قَدْ نَامُوا
> =وَفِي الْجِهَادِ: (سِلْمٌ ضِدُّهُ حَرْبُ)=
> وَ(نَصْرٌ ضِدُّهُ هَزِيمَةٌ) فِي الْقَلْبِ
> =وَفِي الْأَطْعِمَةِ: (طَيِّبٌ ضِدُّهُ خَبِيثُ)=
> وَ(حُلْوٌ ضِدُّهُ مُرٌّ) كَمَا يَقْضِي الْحَدِيثُ

=

= المنظومة السادسة: منظومة في أضداد الصفات والأخلاق

> =تَعَلَّمْ صِفَاتِ النَّفْسِ ضِدّاً وَضِدَّهَا=
> لَعَلَّكَ تَهْتَدِّي إِلَى كُلِّ مَنْهَجِ
> =فَصَبْرٌ ضِدُّهُ جَزَعٌ= وَحِلْمٌ لَهُ
> =غَضَبٌ=، وَرِفْقٌ ضِدُّهُ كُلُّ مُزْعِجِ
> =وَعِلْمٌ ضِدُّهُ جَهْلٌ=، وَصِدْقٌ لَهُ
> =كَذِبٌ=، وَأَمْنٌ ضِدُّهُ كُلُّ مُحْرِجِ
> =وَعَدْلٌ ضِدُّهُ ظُلْمٌ=، وَعَفْوٌ لَهُ
> =انْتِقَامٌ=، وَإِنْصَافٌ ضِدُّ التَّحَزُّجِ
> =وَبِرٌّ ضِدُّهُ عُقُوقٌ=، وَإِحْسَا
> نٌ ضِدُّهُ إِسَاءَةٌ لَمْ تُعَرَّجِ

=

> =وَجُودٌ ضِدُّهُ بُخْلٌ=، وَحَمْدٌ لَهُ
> =ذَمٌّ=، وَتَوْحِيدٌ ضِدُّ التَّشَرُّجِ
> =وَلِينٌ ضِدُّهُ خُرْقٌ=، وَسَلْمٌ لَهُ
> =حَرْبٌ=، وَتَقْوَى ضِدُّ كُلِّ مُدَجَّجِ
> =وَعِفَّةٌ ضِدُّهَا فُجُورٌ=، وَشُكْرٌ
> ضِدُّهُ =كُفْرَانُ= النَّعِيمِ الْمُبَهَجِ
> =وَصِحَّةٌ ضِدُّهَا سَقَامٌ=، وَعُمْرٌ
> ضِدُّهُ =مَوْتٌ=، وَشَبَابٌ لَهُ هَرَمٌ يَلُجُّ
> =فَخُذْ مِنْ صِفَاتِ الْخَيْرِ مَا شِئْتَ وَاَبْتَعِدْ=
> عَنِ الشَّرِّ فَالأَضْدَادُ تُهْدِي إِلَى الْفَرَجِ

=

= المنظومة السابعة: منظومة في أضداد الطبيعة والأزمنة

> =وَلِلْأَرْضِ أَضْدَادٌ تُعَدُّ كَثِيرَةٌ=
> تَأَمَّلْهَا تَظْفَرْ بِحُسْنِ صَنِيعِ
> =فَبَرٌّ ضِدُّهُ بَحْرٌ=، وَسَهْلٌ لَهُ
> =جَبَلٌ=، وَشَرْقٌ ضِدُّهُ كُلُّ مُرْبِعِ
> =وَحَرٌّ ضِدُّهُ بَرْدٌ=، وَصَيْفٌ لَهُ
> =شِتَاءٌ=، وَخِصْبٌ ضِدُّهُ كُلُّ مُجْدِبِ
> =وَنُورٌ ضِدُّهُ ظُلْمَةٌ=، وَضُحًى لَهُ
> =دُجًى=، وَصَبَاحٌ ضِدُّ لَيْلٍ مُعَتَّبِ
> =وَمَاءٌ ضِدُّهُ نَارٌ=، وَغَيْثٌ لَهُ
> =جَفَافٌ=، وَرَوْضٌ ضِدُّ قَفْرٍ مُسَبِّبِ

=

> =وَرَبْعٌ ضِدُّهُ خَرَابٌ=، وَعُمْرَا
> نٌ ضِدُّهُ تَهْدِيمُ كُلِّ مُشَيَّدِ
> =وَقُرْبٌ ضِدُّهُ بُعْدٌ=، وَدَانٍ لَهُ
> =شَاسِعٌ=، وَيَقْظَانٌ ضِدُّ مُهَوَّدِ
> =وَحَيٌّ ضِدُّهُ مَيْتٌ=، وَبَاقٍ لَهُ
> =فَانٍ=، وَأَوَّلُ ضِدُّهُ كُلُّ مُبْعِدِ
> =فَسُبْحَانَ مَنْ سَوَّى الْأَضْدَادَ كُلَّهَا=
> وَأَتْقَنَهَا فِي صُنْعِهِ كُلَّ مُتْقِنِ

=

= المنظومة الثامنة: منظومة قصيرة في أضداد الحروف والمعاني النحوية

> =وَفِي النَّحْوِ أَضْدَادٌ لِمَنْ كَانَ طَالِباً=
> يَحُصِّلُ عِلْماً نَافِعاً كُلَّ مَطْلَبِ
> =فَرَفْعٌ ضِدُّهُ نَصْبٌ=، وَجَزْمٌ لَهُ
> =جَرٌّ=، وَإِعْرَابٌ ضِدُّهُ كُلُّ مُعْرِبِ
> =وَمَاضٍ ضِدُّهُ مُسْتَقْبَلٌ=، وَمُذَكَّرٌ
> لَهُ =مُؤَنَّثٌ=، وَاسْمٌ ضِدُّ مُرَكَّبِ
> =وَمُفْرَدٌ ضِدُّهُ جَمْعٌ=، وَمُظْهَرٌ لَهُ
> =مُضْمَرٌ=، وَتَعْرِيفٌ ضِدُّ مُنَكَّبِ
> =وَمَبْنِيٌّ ضِدُّهُ مُعْرَبٌ=، وَمُتَّصِلٌ
> ضِدُّهُ =مُنْفَصِلٌ= فِي الْقَوْلِ فَاِضْبِطْ وَرَتِّبِ

=

= المنظومة التاسعة: منظومة في أضداد الألفاظ القرآنية

> =وَفِي الذِّكْرِ أَلْفَاظٌ تَضُمُّ الْأَضَادَّا=
> تَدَبَّرْهَا تَزْدَدْ مِنَ الْعِلْمِ أَضْعَافاً
> =فَمِنْهَا: (اشْتَرَوُا) الضَّلَالَةَ بِالْهُدَى=
> فَكَانَ اشْتِرَاهُمْ بَيْعَهُمْ فِيهِ إِنْصَافاً
> =وَمِنْهَا: (الظَّنُّ) فِي الْيَقِينِ وَفِي الشَّكِّ=
> كِلَاهُمَا قَدْ جَاءَ فِي النَّصِّ أَوْصَافاً
> =وَمِنْهَا: (وَرَاءَ) لِلْأَمَامِ وَلِلْخَلْفِ=
> كَذَاكَ قَدْ فَسَّرَهُ الْعُلَمَاءُ إِنْصَافاً
> =وَمِنْهَا: (الْقُرْءُ) فِي الطُّهْرِ وَالْحَيْضِ=
> وَ(الْعَسْعَسَةُ) لِلْإِقْبَالِ وَالْإِدْبَارِ أَضْعَافاً

=

= المنظومة العاشرة: خاتمة جامعة في الأضداد

> =هَذِي الْأَضَادُّ جَمَعْتُهَا فِي مَنْظِمٍ=
> لِتَكُونَ تَذْكَاراً لِكُلِّ لَبِيبِ
> =فَخُذِ الْعُلُومَ مِنَ الْأَوَائِلِ إِنَّهَا=
> نِعْمَ الدَّلِيلُ لِطَالِبٍ وَمُصِيبِ
> =وَالْأَضْدَادُ فِي الْقُرْآنِ وَالشِّعْرِ الَّذِي=
> قَدْ جَاءَ عَنْ عَرَبٍ هُمُ الْأَعْرَابِ
> =فَاحْفَظْهَا تَنْجُ مِنَ الْجَهَالَةِ وَاْرْتَقِ=
> فِي دَرَجِ الْعِرْفَانِ كُلَّ صَعِيبِ
> =وَاللَّهُ أَسْأَلُ أَنْ يُتِمَّ فُهُومَنَا=
> وَيَقِيَنَا مِنْ زَلَّةٍ وَعُيُوبِ

=

هل تريد بعدها الانتقال إلى =أبواب أخرى من الشعر العربي=، كالشعر التعليمي في التجويد والقراءات، أو العودة إلى =توسيع أي باب مما سبق=؟
